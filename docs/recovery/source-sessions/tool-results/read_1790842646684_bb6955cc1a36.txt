   118→  days: number,
   119→  startDate?: string

   121→  if (!startDate) return [];

   123→  const anchors: AnchorForecast[] = [];
   124→  await Promise.all(
   125→    WEATHER_ANCHOR_DEFS.map(async (def) => {
   126→      const dest = DESTINATIONS.find((d) => d.id === def.id);
   127→      if (!dest) return;
   128→      const forecast = await fetchDailyForecast(
   129→        dest.coords.lat,
   130→        dest.coords.lng,
   131→        days,
   132→        startDate
   133→      );
   134→      if (forecast) {
   135→        anchors.push({ label: def.label, labelEn: def.labelEn, forecast });
   136→      }
   137→    })
   138→  );
   139→  return anchors;

   522→//

   524→// Po generiranju (AI ali fallback) zato itinerer obogatimo z realno DAILY

   531→//
   532→// WEATHER-CONTEXT: napoved je zdaj poravnana z datumom odhoda (startDate —
   533→// prej je dan i dobil "i-ti dan od danes", tudi za potovanje čez teden) in

   535→// zgodi prej, v promptu (glej fetchAnchorForecasts zgoraj).
   536→async function enrichWithRealWeather(
   537→  itinerary: Itinerary,
   538→  startDate?: string,
   539→  lang: PlannerLang = "sl"
   540→): Promise<Itinerary> {
   541→  try {
   542→    const firstLoc = itinerary.days[0]?.locations?.[0];
   543→    if (!firstLoc?.destination_id) return itinerary;
   544→
   545→    const dest = DESTINATIONS.find((d) => d.id === firstLoc.destination_id);
   546→    if (!dest) return itinerary;
   547→
   548→    const daily = await fetchDailyForecast(
   549→      dest.coords.lat,
   550→      dest.coords.lng,
   551→      itinerary.days.length,
   552→      startDate
   553→    );
   554→    if (!daily) {
   555→      // TASK 4 / K-2 (UX FIX PASS): Open-Meteo ni dosegljiv — obstoječe
   556→      // vreme (AI-izmišljeno ali sezonska ocena iz determinističnega motorja)
   557→      // NI realna napoved, zato ga izrecno označimo kot oceno. TrustLine
   558→      // zaradi tega NE izriše "✓ Vreme preverjeno" (živi dokaz audita:
   559→      // Render /api/weather → error, itinerer pa je trdil "sončno 22°" —
   560→      // realno je bilo megla/nevhta). ISKRENOST > lep prikaz.
   561→      return {
   562→        ...itinerary,
   563→        days: itinerary.days.map((d) => ({ ...d, weatherEstimated: true })),
   564→      };
   565→    }
   566→
   567→    const tips = Array.isArray(itinerary.tips) ? [...itinerary.tips] : [];
   568→
   569→    for (let i = 0; i < itinerary.days.length; i++) {
   570→      const forecast = daily[i];
   571→      // Če prognoza nima dneva i (krajša od itinererja), obdrži obstoječe vreme
   572→      // — a označi kot oceno (tega dne napoved NI bila preverjena).
   573→      if (!forecast) {
   574→        itinerary.days[i] = { ...itinerary.days[i], weatherEstimated: true };
   575→        continue;
   576→      }
   577→
   578→      itinerary.days[i] = {
   579→        ...itinerary.days[i],
   580→        weather: {
   581→          // W1-faza-2b: 4-jezični dispečer (weatherCodeToTextFor)
   582→          condition: weatherCodeToTextFor(lang, forecast.weatherCode),
   583→          temp: Math.round(forecast.tempMax),
   584→        },
   585→        // TASK 4 / K-2: realna Open-Meteo napoved — izrecni marker za
   586→        // TrustLine (✓ sme se izrisati SAMO nad tem).
   587→        weatherEstimated: false,
   588→      };
   589→
   590→      // Dež alternative — dodaj v tips, če je verjetnost padavin visoka
   591→      // (P4-8: EN uporabnik dobi EN tip; W1-faza-2b: IT/DE različice;
   592→      // W12-faza-2b: FR/ES različice — ista kanonska destinacija imena)
   593→      if ((forecast.precipitationProbabilityMax ?? 0) >= 60) {
   594→        const tip = PL(lang, {
   595→          sl: `Dan ${i + 1}: verjeten dež — alternative: Postojnska/Škocjanske jame, muzeji, terme Terme Olimia.`,
   596→          en: `Day ${i + 1}: rain likely — alternatives: Postojna/Škocjan Caves, museums, Terme Olimia thermal spa.`,
   597→          it: `Giorno ${i + 1}: pioggia probabile — alternative: Grotte di Postumia/Škocjan, musei, terme di Terme Olimia.`,
   598→          de: `Tag ${i + 1}: Regen wahrscheinlich — Alternativen: Postojna-/Škocjan-Höhlen, Museen, Therme Terme Olimia.`,
   599→          fr: `Jour ${i + 1} : pluie probable — alternatives : grottes de Postojna/Škocjan, musées, thermes de Terme Olimia.`,
   600→          es: `Día ${i + 1}: lluvia probable — alternativas: cuevas de Postojna/Škocjan, museos, Terme Olimia.`,
   601→        });
   602→        if (!tips.includes(tip)) tips.push(tip);
   603→      }
   604→    }
   605→
   606→    itinerary.tips = tips;
   607→    return itinerary;
   608→  } catch (error) {
   609→    console.error("[itinerary] enrichWithRealWeather napaka:", error);
   610→    return itinerary;
   611→  }
   612→}
