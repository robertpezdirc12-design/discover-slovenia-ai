   323→/**

   330→ *  - postanek BREZ geo → gumba NI (iskrena odsotnost — kot razdalja);

   332→ */
   333→function NavButton({
   334→  card,
   335→  lang,
   336→  variant = "hero",
   337→  origin,
   338→}: {
   339→  card: GoEntryCard;
   340→  lang: "sl" | "en";
   341→  variant?: "hero" | "icon";
   342→  /** ISSUE #4 §8: živi GPS — prenese se v web URL (external app dobi
   343→   * dejansko izhodišče; brez njega uporabi svojo lokacijo). */
   344→  origin?: { lat: number; lng: number } | null;
   345→}) {
   346→  const links = goNavLinks(card.entry, origin ?? null);
   347→  if (links == null) return null; // brez geo → handoff preprosto NI
   348→
   349→  const onNav = (e: React.MouseEvent<HTMLAnchorElement>) => {
   350→    // Mobilni: geo: URI odpre izbirnik aplikacij (precej nad web URL — brez
   351→    // privzganja Google Maps). Prestrežemo SAMO ob kliku (0 hidracijskih
   352→    // posledic). Desktop/pad: privzeti <a> odpre Google Maps.
   353→    if (isCoarsePointer()) {
   354→      e.preventDefault();
   355→      window.location.href = links.geo;
   356→    }
   357→  };
   358→
   359→  const externalHint = GO_NAV_LABELS.external[lang];
   360→  const aria = GO_NAV_LABELS.navigateAria[lang](card.entry.title);
   361→
   362→  if (variant === "icon") {
   363→    return (
   364→      <Button asChild variant="outline" className="h-11 w-11 shrink-0">
   365→        <a
   366→          href={links.web}
   367→          target="_blank"
   368→          rel="noopener noreferrer"
   369→          onClick={onNav}
   370→          title={externalHint}
   371→          aria-label={aria}
   372→        >
   373→          <NavigationIcon className="h-4 w-4" />
   374→        </a>
   375→      </Button>
   376→    );
   377→  }
   378→
   379→  return (
   380→    <Button
   381→      asChild
   382→      variant="outline"
   383→      className="h-12 flex-1 border-emerald-300 text-emerald-800 hover:bg-emerald-50 hover:text-emerald-900 dark:border-emerald-800 dark:text-emerald-300 dark:hover:bg-emerald-950"
   384→    >
   385→      <a
   386→        href={links.web}
   387→        target="_blank"
   388→        rel="noopener noreferrer"
   389→        onClick={onNav}
   390→        title={externalHint}
   391→        aria-label={aria}
   392→      >
   393→        <NavigationIcon className="mr-2 h-4 w-4" /> {GO_NAV_LABELS.navigate[lang]}
   394→      </a>
   395→    </Button>
   396→  );
   397→}

   403→export function GoMode() {

   412→  const [done, setDone] = useState<Record<string, string>>({});
   413→  const [now, setNow] = useState<Date | null>(null);
   414→  const [showDone, setShowDone] = useState(false);
   415→  // ISSUE #4 §16 (val 4): DNEVNA NAVIGACIJA — ročno izbrani dan (null =
   416→  // samodejno po datumu). Preživi tick ure/geo — uporabnikova izbira je
   417→  // stabilna, dokler jo ne resetira (gumb »Danes«).
   418→  const [dayOverride, setDayOverride] = useState<number | null>(null);
   419→  const geo = useGeolocation();
   420→
   421→  useEffect(() => {
   422→    const hydrate = setTimeout(() => {
   423→      setRecord(loadGoTrip());
   424→      setDone(loadGoProgress());

   428→    // zunanji dogodek, ne sinhroni render kaskada).

   430→    return () => {
   431→      clearTimeout(hydrate);

   434→  }, []);
   435→
   436→  const trip = useMemo(

   444→          : record.view
   445→        : null,
   446→    [record]
   447→  );
   448→  const view = useMemo(
   449→    () =>
   450→      trip && now
   451→        ? buildGoView(trip, now, geo.position, done, { dayOverride })
   452→        : null,

   454→  );
   455→
   456→  // ----------------------------------------------------------------------
   457→  // TASK 65 — VREME PRI NASLEDNJI POSTANKI (živi Open-Meteo prek
   458→  // /api/weather, 10-min cache na strežniku). Cilj je GEO naslednjega
   459→  // postanka — postanek brez geo → vreme preprosto NI (iskrena odsotnost,
   460→  // isti kanon kot DistanceChip). Pogled se gradi vsakih 30 s (živa ura) —
   461→  // zato so effect-depi PRIMITIVI (lat/lng), da se fetch sproži SAMO ob
   462→  // spremembi postanka (ne ob vsakem tiku ure).
   463→  // ----------------------------------------------------------------------
   464→  const weatherTarget = useMemo(
   465→    () => (view ? goWeatherTarget(view) : null),
   466→    [view]
   467→  );
   468→  const wLat = weatherTarget?.lat ?? null;
   469→  const wLng = weatherTarget?.lng ?? null;
   470→  const [weather, setWeather] = useState<GoWeather | null>(null);
   471→  const [weatherFailed, setWeatherFailed] = useState(false);
   472→  /** Za kateri cilj je trenutni odgovor veljalen (drugo = zastarel → skeleton). */
   473→  const [weatherFor, setWeatherFor] = useState<{
