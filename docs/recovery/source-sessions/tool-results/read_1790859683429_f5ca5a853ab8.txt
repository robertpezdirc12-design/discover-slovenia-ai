   327→  // aritmetika je DST-varna.
   328→  var MONTHS_GEN = ["januarja", "februarja", "marca", "aprilja", "maja",
   329→    "junija", "julija", "avgusta", "septembra", "oktobra", "novembra", "decembra"];

   332→  function parseLocalDate(v) {
   333→    if (typeof v !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(v)) return null;
   334→    var p = v.split("-");
   335→    var y = +p[0], mo = +p[1], d = +p[2];
   336→    var dt = new Date(y, mo - 1, d);
   337→    if (dt.getFullYear() !== y || dt.getMonth() !== mo - 1 || dt.getDate() !== d) {
   338→      return null; // roll-over (npr. 2026-02-31) je neveljaven datum
   339→    }
   340→    return { y: y, mo: mo, d: d };
   341→  }

   575→      if (!raw) return null;
   576→      var rec = JSON.parse(raw);
   577→      if (!rec || typeof rec !== "object" || Array.isArray(rec)) return null;
   578→      if (typeof rec.savedAt !== "string") return null;
   579→      if (rec.version === 1) {
   580→        var j = rec.journey;
   581→        if (!j || typeof j !== "object") return null;
   582→        if (typeof j.id !== "string" || (j.lang !== "sl" && j.lang !== "en")) return null;
   583→        if (!j.origin || !j.destination ||
   584→            typeof j.origin.label !== "string" ||
   585→            typeof j.destination.label !== "string") return null;
   586→        if (!j.categories || typeof j.categories !== "object") return null;
   587→        if (!Array.isArray(rec.selectedIds) ||
   588→            !rec.selectedIds.every(function (id) { return typeof id === "string"; })) {
   589→          return null;
   590→        }
   591→        return rec;
   592→      }
   593→      if (rec.version === 2) {
   594→        if (rec.kind !== "itinerary") return null;
   595→        var view = rec.view;
   596→        if (!view || typeof view !== "object") return null;
   597→        if (!Array.isArray(view.days)) return null;

   600→            d.entries.every(function (e) {
   601→              return e && typeof e === "object" && typeof e.title === "string";
   602→            });
   603→        });
   604→        if (!okDays) return null;

   616→  function goTimeline(rec) {
   617→    // ISSUE #4 §16 (val 4): V2 — AI itinerer (MyTripView.days). Izris iz
   618→    // ISTIH polj kot GoMode: naslovi vnosa + SAMO realni časi (time.start),
   619→  // 0 izumljenih ur/cen. dateLabel je dvojezičen vzame [LANG].
   620→    if (rec.version === 2) {

   627→          var t = e.title;
   628→          if (e.time && typeof e.time.start === "string") {
   629→            t = e.time.start.slice(0, 5) + " · " + t;
   630→          }

   633→        stops2 += titles.length;

   635→          date: typeof d.date === "string" ? d.date : "",
   636→          label: d.dateLabel && typeof d.dateLabel[LANG] === "string"
   637→            ? d.dateLabel[LANG] : (typeof d.date === "string" ? fmtDate(d.date) : T.goNoDate),
   638→          titles: titles
   639→        });
   640→      });
   641→      return { days: days2, stops: stops2 };
   642→    }
   643→    var j = rec.journey;
   644→    var sel = {};

   646→    function picked(catKey) {

   649→      return prods.filter(function (p) {

   651→          typeof p.title === "string" && sel[p.id] === true;
   652→      });
   653→    }
   654→    var day1 = [(j.lang === "en" ? "Arrival: " : "Prihod: ") + j.origin.label];

   656→      function (k) { picked(k).forEach(function (p) { day1.push(p.title); }); }
   657→    );
   658→    var evByDate = {}, evOrder = [];

   661→        ? p.eventDate.start.slice(0, 10) : null;
   662→      if (!d) return;

   669→      titles: day1

   674→    days.sort(function (a, b) { return a.date.localeCompare(b.date); });

   677→    return { days: days, stops: stops };
