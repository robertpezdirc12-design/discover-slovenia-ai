   214→    );

   217→    <Badge className="bg-emerald-100 text-emerald-900 border-emerald-300 hover:bg-emerald-100">

   234→export function JourneyPlanner() {
   235→  const locale = useLocale();
   236→  const lang: "sl" | "en" = locale === "en" ? "en" : "sl";
   237→  const router = useRouter();

   241→  const { items: myTripItems } = useMyTrip();
   242→
   243→  const [origin, setOrigin] = useState("Brnik");

   245→  const [startDate, setStartDate] = useState("");
   246→  const [arrivalTime, setArrivalTime] = useState("14:00");
   247→  const [travelers, setTravelers] = useState(2);
   248→  const [cats, setCats] = useState<JourneyCategoryKey[]>(ALL_CATS);
   249→  const [loading, setLoading] = useState(false);
   250→  const [error, setError] = useState<string | null>(null);
   251→  const [journey, setJourney] = useState<TravelJourney | null>(null);
   252→  const [selected, setSelected] = useState<Set<string>>(new Set());
   253→  // §21: tiskalniški pogovor pokaže SAMO potrditveni dokument (MY TRIP).
   254→  const [printMode, setPrintMode] = useState(false);
   255→
   256→  const toggleCat = (c: JourneyCategoryKey) =>
   257→    setCats((prev) =>
   258→      prev.includes(c) ? prev.filter((x) => x !== c) : [...prev, c]
   259→    );
   260→
   261→  const plan = useCallback(async () => {
   262→    setLoading(true);
   263→    setError(null);
   264→    try {
   265→      const res = await fetch("/api/journey/plan", {
   266→        method: "POST",
   267→        headers: { "Content-Type": "application/json" },
   268→        body: JSON.stringify({
   269→          origin,
   270→          destination,
   271→          ...(startDate ? { startDate } : {}),
   272→          arrivalTime,
   273→          travelers,
   274→          categories: cats,
   275→          lang,
   276→        }),
   277→      });
   278→      const data = await res.json();
   279→      if (!res.ok) {
   280→        setError(typeof data?.error === "string" ? data.error : "Napaka");
   281→        setJourney(null);
   282→        return;
   283→      }
   284→      setJourney(data.journey as TravelJourney);
   285→      setSelected(new Set());
   286→    } catch {
   287→      setError(lang === "en" ? "Network error" : "Omrežna napaka");
   288→    } finally {
   289→      setLoading(false);
   290→    }
   291→  }, [origin, destination, startDate, arrivalTime, travelers, cats, lang]);
   292→
   293→  const allProducts = useMemo(() => {
   294→    if (!journey) return [];
   295→    const out: JourneyProduct[] = [];
   296→    for (const key of ALL_CATS) {
   297→      out.push(...journey.categories[key]?.products ?? []);
   298→    }
   299→    return out;
   300→  }, [journey]);
   301→
   302→  const selectedProducts = useMemo(
   303→    () => allProducts.filter((p) => selected.has(p.id)),
   304→    [allProducts, selected]
   305→  );
   306→
   307→  // TASK 8 / F3-A — DRIFT A/B popravek (issue #8 §40/§43; 38-a §1f):
   308→  // zbirka "Moja pot" je resnica OGLEDALA izbir za produkte aktualnega
   309→  // potovanja. Dogodki (provider "events") ostajajo session-only (iz
   310→  // zbirke so namenoma izključeni — D8-D). Po novem iskanju se izbire
   311→  // STRUCTURNO rehidrirajo (re-search iste relacije obdrži izbire),
   312→  // odstranitev iz zbirke (npr. drug zavihek /moja-potovanja) pa jih
   313→  // pošteno odstrani tudi tu. Reconcile vrne isto referenco, če ni
   314→  // spremembe (React bailout — brez ploske zanke re-renderjev).
   315→  useEffect(() => {
   316→    setSelected((prev) =>
   317→      journey
   318→        ? new Set(
   319→            reconcileSelectionFromCollection(prev, allProducts, isInMyTrip)

   321→        : prev
   322→    );
   323→  }, [journey, allProducts, myTripItems]);

   362→   */

   365→    saveGoTrip(journey, [...selected]);
   366→    router.push(lang === "en" ? "/en/na-poti" : "/na-poti");

   369→  /**
   370→   * §21: natisni potrditveni dokument — začasno skrij ostale dele strani

   381→      setTimeout(done, 1500);
   382→    });
   383→  }, []);
   384→
   385→  const toggleProduct = (id: string) =>
   386→    setSelected((prev) => {
   387→      const next = new Set(prev);
   388→      if (next.has(id)) next.delete(id);
   389→      else next.add(id);
   390→      return next;
   391→    });
   392→
   393→  const mapProducts = useMemo(
   394→    () =>
   395→      allProducts.map((p) => ({
   396→        ...p,
   397→        mapStatus: selected.has(p.id) ? ("selected" as const) : p.mapStatus,
   398→      })),
   399→    [allProducts, selected]
   400→  );
   401→
   402→  const t = (o: { sl: string; en: string }) => o[lang];
   403→
   404→  return (
   405→    <div className="space-y-6">
   406→      {/* === VHODNI OBRAZEC === */}
   407→      <div className={printMode ? "print:hidden" : ""}>
   408→      <Card>
   409→        <CardHeader>
   410→          <CardTitle className="flex items-center gap-2 text-lg">
   411→            <Plane className="h-5 w-5" /> {t(L.form.title)}
   412→          </CardTitle>
   413→          <p className="text-sm text-muted-foreground">{t(L.form.subtitle)}</p>
   414→          {/* TASK 8 / F3-A (§43): tiha vrstica odnosa — en načrtovalnik,
   415→              dva koraka (ponudniki tukaj, AI načrt tam). */}
   416→          <p className="text-xs leading-snug text-muted-foreground">
   417→            <Compass className="mr-1 inline h-3.5 w-3.5" aria-hidden="true" />
   418→            {t(L.form.note)}
   419→          </p>
   420→        </CardHeader>
   421→        <CardContent className="space-y-4">
   422→          {/* TASK 8 / F3-A: predlogi destinacij "Iz moje poti" — viden SAMO,

   424→              (aria-pressed — trenutna destinacija je poudarjena). */}
   425→          {myTripDestinations.length > 0 && (
   426→            <div className="flex flex-wrap items-center gap-1.5 text-xs">
   427→              <span className="inline-flex shrink-0 items-center gap-1 font-medium text-muted-foreground">
   428→                <Compass className="size-3.5 text-primary" aria-hidden="true" />
   429→                {t(L.form.fromMyTrip)}
   430→              </span>
   431→              {myTripDestinations.map((d) => {
   432→                const active = destination === d.id;
   433→                return (
