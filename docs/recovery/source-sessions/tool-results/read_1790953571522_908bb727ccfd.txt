   161→export function Navigation({ solid = false }: { solid?: boolean }) {
   162→  const [mounted, setMounted] = React.useState(false);

   167→  const [progress, setProgress] = React.useState(0);
   168→  const t = useTranslations("nav");
   169→  const navLocale = useLocale();
   170→  // W1: jezik lupine = aktivni locale (4 javni jeziki; pick() validira).
   171→  const lang = navLocale;
   172→  const router = useRouter();
   173→  const navLinks = useNavLinks();

   177→

   179→  // KLIKI — <SmartSearch> je bil izrisan BREZ onSelectDestination, zato je

   184→  // komponenti). Router je locale-zaveden (@/i18n/navigation), zato EN

   213→

   215→  const glass = scrolled || solid;
   216→
   217→  // Cart store — items prikazujemo šele po mountu, da se izognemo
   218→  // hydration mismatchu (Zustand persist prebere localStorage šele na klientu).
   219→  const cartItems = useCart((s) => s.items);
   220→  const openCart = useCart((s) => s.openCart);
   221→  const cartCount = mounted
   222→    ? cartItems.reduce((sum, i) => sum + i.quantity, 0)
   223→    : 0;
   224→
   225→  React.useEffect(() => setMounted(true), []);
   226→
   227→  // Scroll listener: preklop stekla + progress branja
   228→  React.useEffect(() => {
   229→    let ticking = false;
   230→    const onScroll = () => {
   231→      if (ticking) return;
   232→      ticking = true;
   233→      requestAnimationFrame(() => {
   234→        const y = window.scrollY;
   235→        setScrolled(y > 24);
   236→        const doc = document.documentElement;
   237→        const max = doc.scrollHeight - window.innerHeight;
   238→        setProgress(max > 0 ? Math.min(y / max, 1) : 0);
   239→        ticking = false;
   240→      });
   241→    };
   242→    window.addEventListener("scroll", onScroll, { passive: true });
   243→    onScroll();
   244→    return () => window.removeEventListener("scroll", onScroll);
   245→  }, []);
   246→
   247→  const toggleTheme = () => {
   248→    setTheme(resolvedTheme === "dark" ? "light" : "dark");
   249→  };
   250→
   251→  return (
   252→    <>
   253→      {/* TASK 8 / F2-A: strežniška refleksija zbirke "Moja pot" — nevidni

   256→      <header
   257→        className={cn(
   258→          "sticky top-0 z-[2000] w-full transition-all duration-300",
   259→          glass
   260→            ? "border-b border-border/70 bg-background/85 shadow-[0_4px_24px_-12px_rgba(0,0,0,0.18)] backdrop-blur-xl supports-[backdrop-filter]:bg-background/70"
   261→            : "border-b border-transparent bg-transparent"
   262→        )}
   263→      >
   264→        <div className="mx-auto flex h-16 w-full max-w-7xl items-center justify-between gap-4 px-4 sm:px-6 lg:px-8">
   265→          {/* Logotip */}
   266→          <Link
   267→            href="/"
   268→            className={cn(
   269→              "group flex items-center gap-2 transition-colors",
   270→              glass
   271→                ? "text-foreground hover:text-primary"
   272→                : "text-white drop-shadow-md hover:text-white"
   273→            )}
   274→            aria-label="Discover Slovenia AI — domov"
   275→          >
   276→            <span
   277→              className={cn(
   278→                "flex size-9 items-center justify-center rounded-lg shadow-md transition-all group-hover:scale-105",
   279→                glass
   280→                  ? "bg-primary text-primary-foreground"
   281→                  : "bg-white/15 text-white backdrop-blur-md ring-1 ring-white/30"
   282→              )}
   283→            >
   284→              <Mountain className="size-5" aria-hidden="true" />
   285→            </span>
   286→            {/* Besedna znamka — na zelo ozkih zaslonih (<360px) se skrije,
   287→                ostane gorski znak (prej 2px horizontalni preliv na 320px) */}
   288→            <span className="flex max-[359px]:hidden flex-col leading-none">
   289→              <span className="text-sm font-bold tracking-tight sm:text-base">
   290→                Discover Slovenia AI
   291→              </span>
   292→              <span
   293→                className={cn(
   294→                  "hidden text-[10px] font-medium uppercase tracking-[0.18em] sm:block",
   295→                  glass ? "text-muted-foreground" : "text-white/70"
   296→                )}
   297→              >
   298→                {t("tagline")}
   299→              </span>
   300→            </span>
   301→          </Link>
   302→
   303→          {/* Desktop navigacija — Issue #16 faza 1: Odkrij · Moja pot ·
   304→              Zemljevid · Pojdi + "Več" (dropdown — progressive disclosure) */}
   305→          <nav
   306→            className="hidden items-center gap-1 lg:flex"
   307→            aria-label="Glavna navigacija"
   308→          >
   309→            {navLinks.map((link) => (
   310→              <Link
   311→                key={link.href}
   312→                href={link.href}
   313→                onClick={() => trackShellNav(link.tab, "header")}
   314→                className={cn(
   315→                  "rounded-md px-3 py-2 text-sm font-medium transition-colors",
   316→                  glass
   317→                    ? "text-foreground/80 hover:bg-accent hover:text-accent-foreground"
   318→                    : "text-white/85 hover:bg-white/10 hover:text-white"
   319→                )}
   320→              >
   321→                {link.label}
   322→              </Link>
   323→            ))}
   324→
   325→            {/* Issue #16 §5 — "Več": napredne funkcije se NE odstranijo,
   326→                ampak se umaknejo iz primarne navigacije (progressive
   327→                disclosure). Skupine: Odkrij več · Načrtuj in orodja · Račun. */}

   329→              <DropdownMenuTrigger
   330→                onClick={() => trackShellNav("more", "header")}
   331→                className={cn(
   332→                  "inline-flex items-center gap-1 rounded-md px-3 py-2 text-sm font-medium transition-colors outline-none",

   334→                    ? "text-foreground/80 hover:bg-accent hover:text-accent-foreground focus-visible:ring-2 focus-visible:ring-ring"
   335→                    : "text-white/85 hover:bg-white/10 hover:text-white focus-visible:ring-2 focus-visible:ring-white/60"

   338→                {t("more")}
   339→                <ChevronDown className="size-3.5" aria-hidden="true" />

   343→                {moreLinks.map((link) => (
   344→                  <DropdownMenuItem asChild key={link.href}>
   345→                    <Link
   346→                      href={link.href}
   347→                      onClick={() => trackShellNav("more", "dropdown", link.href)}

   349→                      {link.label}
   350→                    </Link>
