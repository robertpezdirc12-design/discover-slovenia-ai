   107→/** Slovenščina: 1 → ednina, 2 → dvojina, 3–4 → množina, 5+ → splošna množina. */
   108→function slUnit(n: number, one: string, two: string, few: string, many: string): string {
   109→  if (n === 1) return one;
   110→  const r = n % 100;
   111→  if (r === 2) return two;
   112→  if (r === 3 || r === 4) return few;
   113→  return many;
   114→}

   116→/** "pred 2 min", "pred 3 h" — kratke oznake za manj kot uro. */
   117→function slTimeAgo(iso: string, floorIso?: string): string {
   118→  const ms = Date.parse(iso);
   119→  if (Number.isNaN(ms)) return "neznano";
   120→
   121→  // Sanity cap: komentar ne more biti starejši od potovanja samega
   122→  // (pokvarjeni podatki) — v tem primeru pokažemo absolutni datum.
   123→  if (floorIso) {
   124→    const floor = Date.parse(floorIso);
   125→    if (!Number.isNaN(floor) && ms < floor) {
   126→      return new Date(ms).toLocaleDateString("sl-SI", {
   127→        day: "numeric",
   128→        month: "short",
   129→        year: "numeric",
   130→      });
   131→    }
   132→  }
   133→
   134→  const seconds = Math.max(0, Math.floor((Date.now() - ms) / 1000));
   135→  if (seconds < 45) return "pravkar";
   136→
   137→  const minutes = Math.floor(seconds / 60);
   138→  if (minutes < 60) {
   139→    return `pred ${minutes} ${slUnit(minutes, "minuto", "minutama", "minutami", "minutami")}`;
   140→  }
   141→
   142→  const hours = Math.floor(minutes / 60);
   143→  if (hours < 24) {
   144→    return `pred ${hours} ${slUnit(hours, "uro", "urama", "urami", "urami")}`;
   145→  }
   146→
   147→  const days = Math.floor(hours / 24);
   148→  if (days < 30) {
   149→    return `pred ${days} ${slUnit(days, "dnem", "dnevoma", "dnevi", "dnevi")}`;
   150→  }
   151→
   152→  const months = Math.floor(days / 30);
   153→  if (months < 12) {
   154→    return `pred ${months} ${slUnit(months, "mesecem", "mesecema", "meseci", "meseci")}`;
   155→  }
   156→
   157→  const years = Math.floor(days / 365);
   158→  return `pred ${years} ${slUnit(years, "letom", "letoma", "leti", "leti")}`;
   159→}

   161→/** Barva avatarja iz imena (stabilna za isto ime). */
   162→function avatarColor(name: string): string {
   163→  let hash = 0;
   164→  for (const ch of name) hash = (hash * 31 + ch.charCodeAt(0)) >>> 0;
   165→  return AVATAR_COLORS[hash % AVATAR_COLORS.length];
   166→}

   250→export function TripSocial({
   251→  shareId,
   252→  initialComments,
   253→  initialLikes,

   255→}: TripSocialProps) {
   256→  const { toast } = useToast();

   258→  // === Klepet (W2): vrstice kronološko + živost + AI vprašanje ===
   259→  const {
   260→    comments,
   261→    connected,
   262→    aiPending,
   263→    appendLocal,
   264→    emitChatSignal,
   265→    askAi,
   266→  } = useTripChat(shareId, initialComments);
   267→
   268→  // === Stanje — števci se inicializirajo iz server propsov (hidratacija
   269→  // varna), lokalni všeček in clientId se naložita šele v useEffect. ===
   270→  const [likes, setLikes] = useState<number>(() => initialLikes ?? 0);
   271→  const [liked, setLiked] = useState<boolean>(false);
   272→  const [likePending, setLikePending] = useState<boolean>(false);
   273→  const [clientId, setClientId] = useState<string>("");
   274→
   275→  // Obrazec
   276→  const [authorName, setAuthorName] = useState<string>("");
   277→  const [text, setText] = useState<string>("");
   278→  const [submitting, setSubmitting] = useState<boolean>(false);
   279→
   280→  // "Dodaj v pot" — en predlog naenkrat (CAS zaporedno, brez dirk)
   281→  const [addingPlaceId, setAddingPlaceId] = useState<string | null>(null);
   282→  const [addedPlaceIds, setAddedPlaceIds] = useState<ReadonlySet<string>>(
   283→    () => new Set<string>()
   284→  );
   285→
   286→  // Relativni čas računamo šele po mountu (SSR/klient ura se razlikujeta)
   287→  const [mounted, setMounted] = useState<boolean>(false);
   288→
   289→  const textareaRef = useRef<HTMLTextAreaElement | null>(null);
   290→  const scrollRef = useRef<HTMLDivElement | null>(null);
   291→
   292→  // Mount: zagotovi clientId (enak vzorec kot shared-trip glasovanje) +
   293→  // naloži lokalni všeček + prihranjeno ime avtorja
   294→  useEffect(() => {
   295→    setMounted(true);
   296→    if (!shareId) return;
   297→    try {
   298→      // 99-b: read-or-create anonimni ID (private mode → null, stanje
   299→      // ostane nedotaknjeno — isto kot prejšnji try/catch vzorec)
   300→      const cid = getVoterId();
   301→      if (cid) setClientId(cid);
   302→
   303→      setLiked(window.localStorage.getItem(likeStorageKey(shareId)) === "1");
   304→
   305→      const savedName = getAuthorName();
   306→      if (savedName && savedName.trim()) setAuthorName(savedName.trim());
   307→    } catch {
   308→      // localStorage nedostopen (private mode) — všečki/komentarji delujejo
   309→      // brez persistenze lastnega stanja
   310→    }
   311→  }, [shareId]);
   312→
   313→  // === Samodejni drs na dno ob novi vrstici (če je uporabnik že ob dnu —
   314→  // ne smemo mu puliti brskalnika, če bere zgodovino) ===
   315→  const stickToBottomRef = useRef(true);
   316→  useEffect(() => {
   317→    const el = scrollRef.current;
   318→    if (!el) return;
   319→    const onScroll = () => {
   320→      stickToBottomRef.current =
   321→        el.scrollTop + el.clientHeight >= el.scrollHeight - 80;
   322→    };
   323→    el.addEventListener("scroll", onScroll, { passive: true });
   324→    return () => el.removeEventListener("scroll", onScroll);
   325→  }, []);
   326→
   327→  const lastCount = useRef(comments.length);
   328→  useEffect(() => {
   329→    const grew = comments.length > lastCount.current;
   330→    lastCount.current = comments.length;
   331→    if (!grew) return;
   332→    if (!stickToBottomRef.current) return; // uporabnik bere višje — ne puli
   333→    const el = scrollRef.current;
   334→    if (el) el.scrollTop = el.scrollHeight;
   335→  }, [comments.length, aiPending]);
   336→
   337→  // === Všeček — toggle z optimističnim UI in revertom ob napaki ===
   338→  const toggleLike = useCallback(async () => {
   339→    if (!shareId || !clientId || likePending) return;
   340→
   341→    const prevLiked = liked;
   342→    const prevLikes = likes;
   343→    const nextLiked = !prevLiked;
   344→
   345→    // Optimistična posodobitev UI
   346→    setLikePending(true);
   347→    setLiked(nextLiked);
   348→    setLikes((n) => Math.max(0, n + (nextLiked ? 1 : -1)));
   349→
   350→    try {
   351→      const res = await fetch("/api/trip-likes", {
   352→        method: "POST",
   353→        headers: { "Content-Type": "application/json" },
   354→        body: JSON.stringify({ shareId, clientId }),
   355→      });
   356→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   357→      const data: unknown = await res.json();
   358→      const d = data as { liked?: unknown; count?: unknown } | null;
   359→
   360→      // Server je avtoriteta
   361→      if (typeof d?.liked === "boolean") setLiked(d.liked);
   362→      if (typeof d?.count === "number" && d.count >= 0) setLikes(d.count);
   363→
   364→      // Persistiraj lokalno stanje všečka
   365→      try {
   366→        if (d?.liked === true) {
   367→          window.localStorage.setItem(likeStorageKey(shareId), "1");
   368→        } else if (d?.liked === false) {
   369→          window.localStorage.removeItem(likeStorageKey(shareId));
   370→        }
   371→      } catch {
   372→        // private mode — ignore
   373→      }
   374→    } catch (err) {
   375→      // Revert optimistične spremembe
   376→      console.error("[trip-social] všeček neuspešen:", err);
   377→      setLiked(prevLiked);
   378→      setLikes(prevLikes);
   379→      toast({
   380→        title: "Všečka ni bilo mogoče shraniti",
   381→        description: "Preveri povezavo in poskusi znova.",
   382→        variant: "destructive",
   383→      });
   384→    } finally {
   385→      setLikePending(false);
   386→    }
   387→  }, [shareId, clientId, likePending, liked, likes, toast]);
   388→
   389→  // === @AI žeton — vstavi na položaj kazalca v vnosno polje ===
   390→  const insertAiMention = useCallback(() => {
   391→    const el = textareaRef.current;
   392→    const value = text;
   393→    const start = el?.selectionStart ?? value.length;
   394→    const end = el?.selectionEnd ?? start;
   395→    const before = value.slice(0, start);
   396→    const after = value.slice(end);
   397→    const needsSpace = before.length > 0 && !/\s$/.test(before);
   398→    const insert = `${needsSpace ? " " : ""}@AI `;
   399→    const next = `${before}${insert}${after}`;
   400→    setText(next);
   401→    // Fokus + kazalec za žetonom (naslednja slika — textarea se še rendra)
   402→    requestAnimationFrame(() => {
   403→      el?.focus();
   404→      const pos = (before + insert).length;
   405→      el?.setSelectionRange(pos, pos);
   406→    });
   407→  }, [text]);
   408→
   409→  // === Objava sporočila (+ morebitno vprašanje @AI svetovalcu) ===
   410→  const submitComment = useCallback(
   411→    async (e: React.FormEvent<HTMLFormElement>) => {
   412→      e.preventDefault();
   413→      if (submitting || !shareId) return;
   414→
   415→      const name = authorName.trim();
   416→      const body = text.trim();
   417→      const wantsAi = isAiMention(body);
   418→
   419→      if (name.length < 1 || name.length > AUTHOR_NAME_MAX) {
   420→        toast({
   421→          title: "Vpiši svoje ime",
   422→          description: `Ime mora imeti med 1 in ${AUTHOR_NAME_MAX} znakov.`,
   423→          variant: "destructive",
   424→        });
   425→        return;
   426→      }
   427→      if (body.length < 2 || body.length > COMMENT_TEXT_MAX) {
   428→        toast({
   429→          title: "Sporočilo je prekratko ali predolgo",
   430→          description: `Sporočilo mora imeti med 2 in ${COMMENT_TEXT_MAX} znakov.`,
   431→          variant: "destructive",
   432→        });
   433→        return;
   434→      }
   435→
   436→      setSubmitting(true);
   437→      stickToBottomRef.current = true; // svoja vrstica → drs na dno
   438→      try {
   439→        // 1. Sporočilo se vedno objavi (vidno vsem — tudi če AI kasneje
   440→        // odpove; iskrenost: napaka AI ne poje uporabnikovega sporočila).
   441→        const res = await fetch("/api/trip-comments", {
   442→          method: "POST",
   443→          headers: { "Content-Type": "application/json" },
   444→          body: JSON.stringify({ shareId, authorName: name, text: body }),
   445→        });
   446→        const data: unknown = await res.json().catch(() => null);
   447→        const d = data as { success?: unknown; comment?: unknown; error?: unknown } | null;
   448→
   449→        if (!res.ok || d?.success !== true || !d?.comment) {
   450→          const error = typeof d?.error === "string" ? d.error : null;
   451→          throw new Error(error ?? `HTTP ${res.status}`);
   452→        }
   453→
   454→        const comment = d.comment as TripChatItem;
   455→        appendLocal({
   456→          id: comment.id,
   457→          authorName: comment.authorName,
   458→          text: comment.text,
   459→          isAI: comment.isAI === true,
   460→          payload: comment.payload ?? null,
   461→          createdAt: comment.createdAt,
   462→        });
   463→        emitChatSignal(comment.id);
   464→        setText("");
   465→
   466→        // Prihrani ime za naslednje sporočilo (defenzivno — private mode
   467→        // se mirno preskoči znotraj lib-a)
   468→        saveAuthorName(name);
   469→
   470→        if (!wantsAi) {
   471→          toast({
   472→            title: "Sporočilo je objavljeno",
   473→            description: "Hvala, da deliš mnenje s skupino!",
   474→          });
   475→          return;
   476→        }
   477→
   478→        // 2. Vprašanje @AI — strežniški odgovor (isti pogon kot /api/chat).
   479→        // Indikator tipkanja je LOKALEN (aiPending) — prisotni vidijo
   480→        // vrstico prek svojega pollinga/chat:new.
   481→        trackPlannerEvent("chat_group_ai_asked", {
   482→          question_len: body.length,
   483→          surface: "trip-chat",
   484→        });
   485→        try {
   486→          await askAi(body);
   487→          // Odgovor se doda prek appendLocal znotraj askAi — mehurček je
   488→          // povratna informacija, toast bi bil odvečen.
   489→        } catch (err) {
   490→          const message =
   491→            err instanceof Error ? err.message : "Poskusi znova.";
   492→          toast({
   493→            title: "AI svetovalec trenutno ne more odgovoriti",
   494→            description: message,
   495→            variant: "destructive",
   496→          });
   497→        }
   498→      } catch (err) {
   499→        const message =
