   112→function errText(e: unknown): string {
   113→  if (e instanceof Error) return e.message;
   114→  return String(e);

   116→
   117→async function apiJson(
   118→  url: string,
   119→  init: RequestInit & { editToken?: string | null }
   120→): Promise<Response> {
   121→  const { editToken, ...rest } = init;
   122→  const headers = new Headers(rest.headers);
   123→  if (editToken) headers.set("x-dsa-edit-token", editToken);
   124→  if (rest.body && !headers.has("content-type")) {
   125→    headers.set("content-type", "application/json");
   126→  }
   127→  return fetch(url, { ...rest, headers });
   128→}

   130→export function TripCollaboration({
   131→  shareId,
   132→  initialName,

   134→  shareId: string;
   135→  initialName: string | null;
   136→}) {

   138→  // TASK 8 / F3-B: jezik za L-pattern placeholder nalaganja zgodovine

   142→  const searchParams = useSearchParams();

   145→  const [agg, setAgg] = useState<TripAggregate | null>(null);
   146→  const [loadError, setLoadError] = useState<string | null>(null);

   150→  // Vabilo

   153→  >("idle");
   154→  const [acceptedRole, setAcceptedRole] = useState<string | null>(null);
   155→
   156→  // Upravljanje (lastnik)
   157→  const [inviteRole, setInviteRole] = useState<string>("EDITOR");

   159→  const [inviteUrl, setInviteUrl] = useState<string | null>(null);
   160→  const [inviteNote, setInviteNote] = useState<string | null>(null);
   161→  const [copied, setCopied] = useState(false);
   162→
   163→  // Preimenovanje (CAS)
   164→  const [nameDraft, setNameDraft] = useState(initialName ?? "");
   165→  const [renameState, setRenameState] = useState<"idle" | "saving" | "saved" | "conflict">(
   166→    "idle"

   168→  const nameRef = useRef(initialName ?? "");

   172→  const [revisionsOpen, setRevisionsOpen] = useState(false);
   173→  const [revisions, setRevisions] = useState<RevisionMeta[] | null>(null);
   174→  const [revisionsError, setRevisionsError] = useState<string | null>(null);

   176→  const [restoreState, setRestoreState] = useState<
   177→    "idle" | "done" | "conflict" | "error"
   178→  >("idle");
   179→
   180→  const editToken = useMemo(
   181→    () => (typeof window === "undefined" ? null : getEditToken(shareId)),

   183→  );
   184→
   185→  const load = useCallback(async () => {

   189→      });

   196→      }
   197→      const data = (await r.json()) as TripAggregate;
   198→      setAgg(data);
   199→      setLoadError(null);
   200→    } catch (e) {
   201→      setLoadError(errText(e));
   202→    }
   203→  }, [shareId, editToken]);
   204→
   205→  useEffect(() => {
   206→    void load();
   207→  }, [load]);
   208→
   209→  // ── SPREJEM VABILA ────────────────────────────────────────────────────
   210→  const acceptInvite = useCallback(async () => {
   211→    if (!inviteToken) return;
   212→    setInviteState("accepting");

   214→    try {
   215→      const r = await fetch("/api/trip/collaborators/accept", {
   216→        method: "POST",
   217→        headers: { "content-type": "application/json" },
   218→        body: JSON.stringify({ inviteToken }),
   219→      });
   220→      const data = (await r.json()) as {
   221→        success?: boolean;
   222→        role?: string;
   223→        error?: string;
   224→        alreadyActive?: boolean;
   225→        note?: string;
   226→      };
   227→      if (!r.ok || !data.success) {
   228→        setInviteState("error");
   229→        setActionError(data.error ?? `Napaka ${r.status}`);
   230→        return;
   231→      }
   232→      setInviteState("accepted");
   233→      setAcceptedRole(data.role ?? null);
   234→      if (data.note) setActionError(data.note);
   235→      // Osveži agregat (nova vloga).
   236→      void load();
   237→    } catch (e) {
   238→      setInviteState("error");
   239→      setActionError(errText(e));
   240→    }
   241→  }, [inviteToken, load]);
   242→

   246→    setActionError(null);

   248→    setInviteNote(null);
   249→    try {
   250→      const r = await apiJson(
   251→        `/api/trip/${encodeURIComponent(shareId)}/collaborators`,

   253→          method: "POST",
   254→          editToken,

   256→            role: inviteRole,
   257→            ...(inviteEmail.trim()

   259→              : {}),

   261→        }
   262→      );
   263→      const data = (await r.json()) as {
   264→        success?: boolean;
   265→        inviteUrl?: string;
   266→        emailHint?: string;
   267→        error?: string;
   268→      };
   269→      if (!r.ok || !data.success) {
   270→        setActionError(data.error ?? `Napaka ${r.status}`);
   271→        return;
   272→      }
   273→      setInviteUrl(data.inviteUrl ?? null);
   274→      setInviteNote(data.emailHint ?? null);
   275→      void load();
   276→    } catch (e) {
   277→      setActionError(errText(e));
   278→    } finally {
   279→      setBusy(false);
   280→    }
   281→  }, [shareId, editToken, inviteRole, inviteEmail, load]);
   282→
   283→  // ── SPREMENBA VLOGE / ODVZEM (lastnik) ────────────────────────────────

   285→    async (id: string, patch: { role?: string; status?: string }) => {
   286→      setBusy(true);
   287→      setActionError(null);
   288→      try {
   289→        const r = await apiJson(
   290→          `/api/trip/${encodeURIComponent(shareId)}/collaborators`,
   291→          { method: "PATCH", editToken, body: JSON.stringify({ id, ...patch }) }
   292→        );
   293→        const data = (await r.json()) as { success?: boolean; error?: string };
   294→        if (!r.ok || !data.success) {
   295→          setActionError(data.error ?? `Napaka ${r.status}`);
   296→          return;
   297→        }
   298→        void load();
   299→      } catch (e) {
   300→        setActionError(errText(e));
   301→      } finally {
   302→        setBusy(false);
   303→      }
   304→    },
   305→    [shareId, editToken, load]
   306→  );
   307→
   308→  // ── ZGODOVINA REVIZIJ (§22, leneco nalaganje) ─────────────────────────
   309→  const loadRevisions = useCallback(async () => {
   310→    setRevisionsError(null);
   311→    try {
   312→      const r = await fetchTripRevisions(shareId);
   313→      setRevisions(r.revisions);
   314→    } catch (e) {
   315→      setRevisionsError(errText(e));
   316→    }
   317→  }, [shareId]);
   318→
   319→  const toggleRevisions = useCallback(() => {
   320→    setRevisionsOpen((open) => {
   321→      const next = !open;
   322→      // Leneco: prvo razprtje sproži nalaganje (ne ob vsakem renderju).

   324→        void loadRevisions();
   325→      }
   326→      return next;
   327→    });
   328→  }, [revisions, revisionsError, loadRevisions]);
   329→
   330→  // §22: OBNOVI revizijo = PATCH s prebrano staro vsebino (CAS). Sam
   331→  // obnovitveni zapis NAREDI revizijo trenutne vsebine — zgodovina ostane

   333→  const restoreVersion = useCallback(
   334→    async (version: number) => {
   335→      if (!agg || restoringVersion !== null) return;
   336→      setRestoreState("idle");
   337→      setRestoringVersion(version);
   338→      setActionError(null);
   339→      try {
   340→        const content = await fetchTripRevisionContent(shareId, version);
   341→        const upd = await updateItinerary(
   342→          shareId,
   343→          content,
   344→          agg.version.contentVersion
   345→        );
   346→        // 1.163.2 (#22 REMAINING): uspešna obnovitev (CAS posodobitev) je
   347→        // TUDI shranjevanje — zapiši/osveži vnos v Moja potovanja, da je
   348→        // pot pošteno vidna tudi po zamenjavi naprave/brskalnika in ni
   349→        // izpuščena iz prevzema ob prijavi (strežniško ime je kanon,

   351→        addSavedTrip(shareId, upd.name ?? deriveSavedTripName(content));
   352→        setRestoreState("done");
   353→        // RSC stran /pot/[shareId] — celotna osvežitev pobere svežo
   354→        // vsebino + novo verzijo (pošteno in preprosto).
   355→        setTimeout(() => window.location.reload(), 900);
   356→      } catch (e) {
   357→        const msg = errText(e);
   358→        if (msg.includes("sočasno urejanje")) {
   359→          setRestoreState("conflict");
   360→        } else {
   361→          setRestoreState("error");
   362→          setActionError(msg);
   363→        }
   364→      } finally {
   365→        setRestoringVersion(null);
   366→      }
   367→    },
   368→    [shareId, agg, restoringVersion]
   369→  );
   370→
   371→  // ── JAVNA/ZASEBNA POVEZAVA (lastnik) ──────────────────────────────────
   372→  const togglePublic = useCallback(
   373→    async (next: boolean) => {
   374→      setBusy(true);
   375→      setActionError(null);
   376→      try {
   377→        const r = await apiJson(`/api/trip/${encodeURIComponent(shareId)}`, {
   378→          method: "PATCH",
   379→          editToken,
   380→          body: JSON.stringify({ isPublic: next }),
   381→        });
   382→        const data = (await r.json()) as {
   383→          success?: boolean;
   384→          error?: string;
   385→        };
   386→        if (!r.ok || !data.success) {
   387→          setActionError(data.error ?? `Napaka ${r.status}`);
   388→          return;
   389→        }
   390→        void load();
   391→      } catch (e) {
   392→        setActionError(errText(e));
   393→      } finally {
   394→        setBusy(false);
   395→      }
   396→    },
   397→    [shareId, editToken, load]

   402→    if (!agg) return;
   403→    const trimmed = nameDraft.trim();
   404→    if (!trimmed || trimmed === nameRef.current) {
   405→      setRenameState(trimmed === nameRef.current ? "saved" : "idle");

   411→      const r = await apiJson(
   412→        `/api/itinerary/shared/${encodeURIComponent(shareId)}`,
   413→        {
   414→          method: "PATCH",
   415→          editToken,
   416→          body: JSON.stringify({
   417→            baseVersion: agg.version.contentVersion,
   418→            name: trimmed,
   419→          }),
   420→        }
   421→      );
   422→      const data = (await r.json()) as {
   423→        success?: boolean;
   424→        contentVersion?: number;
   425→        error?: string;
   426→        conflict?: boolean;
   427→      };
   428→      if (r.status === 409 || data.conflict) {
   429→        setRenameState("conflict");
   430→        return;
   431→      }
   432→      if (!r.ok || !data.success) {
   433→        setRenameState("idle");
   434→        setActionError(data.error ?? `Napaka ${r.status}`);
   435→        return;
   436→      }
   437→      nameRef.current = trimmed;
   438→      setRenameState("saved");
   439→      void load();
   440→    } catch (e) {
   441→      setRenameState("idle");
   442→      setActionError(errText(e));
   443→    }
   444→  }, [agg, nameDraft, shareId, editToken, load]);
   445→
   446→  const copyInvite = useCallback(async () => {
   447→    if (!inviteUrl) return;
   448→    try {
   449→      await navigator.clipboard.writeText(
   450→        `${window.location.origin}${inviteUrl}`
   451→      );
   452→      setCopied(true);
   453→      setTimeout(() => setCopied(false), 2000);
   454→    } catch {
   455→      setActionError("Kopiranje ni uspelo — povezavo prepiši ročno.");
   456→    }
   457→  }, [inviteUrl]);
   458→
   459→  const role = agg?.role ?? "NONE";
   460→  const showPanel =
   461→    role === "OWNER" || role === "EDITOR" || inviteToken != null;
   462→
   463→  if (loadError) {
   464→    return (
   465→      <Card>
   466→        <CardContent className="flex items-center gap-2 p-4 text-sm text-muted-foreground">
   467→          <AlertTriangle className="h-4 w-4 shrink-0" aria-hidden="true" />
   468→          <span>Sodelovanje trenutno ni na voljo: {loadError}</span>
   469→        </CardContent>
   470→      </Card>
   471→    );
   472→  }
   473→
   474→  if (!showPanel && inviteState === "idle") return null;
   475→
   476→  const loggedIn =
   477→    sessionStatus === "authenticated" &&
   478→    session?.user?.accountType === "user";
   479→
   480→  return (
   481→    <div className="space-y-4">
   482→      {/* ── TRAK VABILA (?invite=…) ──────────────────────────────────── */}
   483→      {inviteToken && (
   484→        <Card className="border-violet-300 ring-1 ring-violet-300/50">
   485→          <CardContent className="space-y-3 p-4">
   486→            <div className="flex items-center gap-2">
   487→              <UserPlus className="h-4 w-4 text-violet-700 dark:text-violet-400" aria-hidden="true" />
   488→              <p className="text-sm font-semibold">
   489→                Povabljen si k sodelovanju na tej poti
   490→              </p>
   491→            </div>
   492→            {inviteState === "accepted" ? (
   493→              <p className="flex items-center gap-2 text-sm text-emerald-700 dark:text-emerald-400">
   494→                <ShieldCheck className="h-4 w-4" aria-hidden="true" />
   495→                Vabilo sprejeto — tvoja vloga:{" "}
   496→                <strong>{ROLE_LABELS[acceptedRole ?? ""] ?? acceptedRole}</strong>
   497→              </p>
   498→            ) : !loggedIn ? (
   499→              <p className="text-sm text-muted-foreground">
   500→                Za sprejem vabila se prijavi z računom popotnika — vloga na
   501→                poti je vezana na račun,{" "}
   502→                <a
   503→                  className="font-medium underline underline-offset-2"
   504→                  href={`/prijava?next=/pot/${shareId}%3Finvite%3D${encodeURIComponent(inviteToken)}`}
   505→                >
   506→                  prijavi se
   507→                </a>
   508→                .
   509→              </p>
   510→            ) : (
   511→              <div className="flex flex-wrap items-center gap-3">
   512→                <Button onClick={acceptInvite} disabled={inviteState === "accepting"}>
   513→                  {inviteState === "accepting" ? (
   514→                    <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden="true" />
   515→                  ) : null}
   516→                  Sprejmi vabilo
   517→                </Button>
   518→                <p className="text-xs text-muted-foreground">
   519→                  Vloga se razkrije po sprejemu (klik na povezavo še ni sprejem).
   520→                </p>
   521→              </div>
   522→            )}
   523→            {inviteState === "error" && actionError && (
   524→              <p className="text-sm text-destructive">{actionError}</p>
   525→            )}
   526→          </CardContent>
   527→        </Card>
   528→      )}
   529→
   530→      {/* ── UPRAVLJANJE (lastnik / urednik) ──────────────────────────── */}
   531→      {(role === "OWNER" || role === "EDITOR") && agg && (
   532→        <Card>
   533→          <CardHeader className="pb-3">
   534→            <CardTitle className="flex flex-wrap items-center gap-2 text-base">
   535→              <Users className="h-4 w-4" aria-hidden="true" />
   536→              Sodelovanje na poti
   537→              <Badge variant="secondary">
   538→                {ROLE_LABELS[role] ?? role}
   539→              </Badge>
   540→              {!agg.isPublic && (
   541→                <Badge className="border-amber-300 bg-amber-50 text-amber-900 hover:bg-amber-50 dark:border-amber-800 dark:bg-amber-950 dark:text-amber-200">
   542→                  Zasebna povezava
   543→                </Badge>
   544→              )}
   545→            </CardTitle>
   546→          </CardHeader>
   547→          <CardContent className="space-y-6">
   548→            {actionError && (
   549→              <p className="text-sm text-destructive" role="alert">
   550→                {actionError}
   551→              </p>
