   129→async function apiJson(
   130→  url: string,
   131→  init: RequestInit & { editToken?: string | null }
   132→): Promise<Response> {
   133→  const { editToken, ...rest } = init;
   134→  const headers = new Headers(rest.headers);
   135→  if (editToken) headers.set("x-dsa-edit-token", editToken);
   136→  if (rest.body && !headers.has("content-type")) {
   137→    headers.set("content-type", "application/json");
   138→  }
   139→  return fetch(url, { ...rest, headers });
   140→}
