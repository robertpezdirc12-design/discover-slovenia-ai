   107→   107→/** Slovenščina: 1 → ednina, 2 → dvojina, 3–4 → množina, 5+ → splošna množina. */
   108→   108→function slUnit(n: number, one: string, two: string, few: string, many: string): string {
   109→   109→  if (n === 1) return one;
   110→   110→  const r = n % 100;
   111→   111→  if (r === 2) return two;
   112→   112→  if (r === 3 || r === 4) return few;
   113→   113→  return many;
   114→   114→}

   116→   116→/** "pred 2 min", "pred 3 h" — kratke oznake za manj kot uro. */
   117→   117→function slTimeAgo(iso: string, floorIso?: string): string {
   118→   118→  const ms = Date.parse(iso);
   119→   119→  if (Number.isNaN(ms)) return "neznano";
   120→   120→
   121→   121→  // Sanity cap: komentar ne more biti starejši od potovanja samega
   122→   122→  // (pokvarjeni podatki) — v tem primeru pokažemo absolutni datum.
   123→   123→  if (floorIso) {
   124→   124→    const floor = Date.parse(floorIso);
   125→   125→    if (!Number.isNaN(floor) && ms < floor) {
   126→   126→      return new Date(ms).toLocaleDateString("sl-SI", {
   127→   127→        day: "numeric",
   128→   128→        month: "short",
   129→   129→        year: "numeric",
   130→   130→      });
   131→   131→    }
   132→   132→  }
   133→   133→
   134→   134→  const seconds = Math.max(0, Math.floor((Date.now() - ms) / 1000));
   135→   135→  if (seconds < 45) return "pravkar";
   136→   136→
   137→   137→  const minutes = Math.floor(seconds / 60);
   138→   138→  if (minutes < 60) {
   139→   139→    return `pred ${minutes} ${slUnit(minutes, "minuto", "minutama", "minutami", "minutami")}`;
   140→   140→  }
   141→   141→
   142→   142→  const hours = Math.floor(minutes / 60);
   143→   143→  if (hours < 24) {
   144→   144→    return `pred ${hours} ${slUnit(hours, "uro", "urama", "urami", "urami")}`;
   145→   145→  }
   146→   146→
   147→   147→  const days = Math.floor(hours / 24);
   148→   148→  if (days < 30) {
   149→   149→    return `pred ${days} ${slUnit(days, "dnem", "dnevoma", "dnevi", "dnevi")}`;
   150→   150→  }
   151→   151→
   152→   152→  const months = Math.floor(days / 30);
   153→   153→  if (months < 12) {
   154→   154→    return `pred ${months} ${slUnit(months, "mesecem", "mesecema", "meseci", "meseci")}`;
   155→   155→  }
   156→   156→
   157→   157→  const years = Math.floor(days / 365);
   158→   158→  return `pred ${years} ${slUnit(years, "letom", "letoma", "leti", "leti")}`;
   159→   159→}
