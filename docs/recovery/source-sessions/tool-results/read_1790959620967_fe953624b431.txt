   219→   219→
   220→   220→function slUnit(
   221→   221→  n: number,
   222→   222→  one: string,
   223→   223→  two: string,
   224→   224→  few: string,
   225→   225→  many: string
   226→   226→): string {
   227→   227→  if (n === 1) return one;
   228→   228→  const r = n % 100;
   229→   229→  if (r === 2) return two;
   230→   230→  if (r === 3 || r === 4) return few;
   231→   231→  return many;
   232→   232→}

   234→   234→/** "pred 2 min", "pred 3 h" — relativni čas z varnostnim stropom. */
   235→   235→function slTimeAgo(iso: string, floorIso?: string): string {
   236→   236→  const ms = Date.parse(iso);
   237→   237→  if (Number.isNaN(ms)) return "neznano";
   238→   238→
   239→   239→  // Sanity cap: vpis ne more biti starejši od potovanja
   240→   240→  if (floorIso) {
   241→   241→    const floor = Date.parse(floorIso);
   242→   242→    if (!Number.isNaN(floor) && ms < floor) {
   243→   243→      return new Date(ms).toLocaleDateString("sl-SI", {
   244→   244→        day: "numeric",
   245→   245→        month: "short",
   246→   246→        year: "numeric",
   247→   247→      });
   248→   248→    }
   249→   249→  }
   250→   250→
   251→   251→  const seconds = Math.max(0, Math.floor((Date.now() - ms) / 1000));
   252→   252→  if (seconds < 45) return "pravkar";
   253→   253→
   254→   254→  const minutes = Math.floor(seconds / 60);
   255→   255→  if (minutes < 60) {
   256→   256→    return `pred ${minutes} ${slUnit(minutes, "minuto", "minutama", "minutami", "minutami")}`;
   257→   257→  }
   258→   258→  const hours = Math.floor(minutes / 60);
   259→   259→  if (hours < 24) {
   260→   260→    return `pred ${hours} ${slUnit(hours, "uro", "urama", "urami", "urami")}`;
   261→   261→  }
   262→   262→  const days = Math.floor(hours / 24);
   263→   263→  if (days < 30) {
   264→   264→    return `pred ${days} ${slUnit(days, "dnem", "dnevoma", "dnevi", "dnevi")}`;
   265→   265→  }
   266→   266→  return new Date(ms).toLocaleDateString("sl-SI", {
   267→   267→    day: "numeric",
   268→   268→    month: "short",
   269→   269→    year: "numeric",
   270→   270→  });
   271→   271→}

   273→   273→/** EN dvojnik slTimeAgo ("2 min ago") — isti varnostni strop, "en-GB" datumi. */
   274→   274→function enTimeAgo(iso: string, floorIso?: string): string {
   275→   275→  const ms = Date.parse(iso);
   276→   276→  if (Number.isNaN(ms)) return "unknown";
   277→   277→
   278→   278→  // Sanity cap: vpis ne more biti starejši od potovanja
   279→   279→  if (floorIso) {
   280→   280→    const floor = Date.parse(floorIso);
   281→   281→    if (!Number.isNaN(floor) && ms < floor) {
   282→   282→      return new Date(ms).toLocaleDateString("en-GB", {
   283→   283→        day: "numeric",
   284→   284→        month: "short",
   285→   285→        year: "numeric",
   286→   286→      });
   287→   287→    }
   288→   288→  }
   289→   289→
   290→   290→  const seconds = Math.max(0, Math.floor((Date.now() - ms) / 1000));
   291→   291→  if (seconds < 45) return "just now";
   292→   292→
   293→   293→  const minutes = Math.floor(seconds / 60);
   294→   294→  if (minutes < 60) {
   295→   295→    return `${minutes} min ago`;
   296→   296→  }
   297→   297→  const hours = Math.floor(minutes / 60);
   298→   298→  if (hours < 24) {
   299→   299→    return `${hours} h ago`;
   300→   300→  }
   301→   301→  const days = Math.floor(hours / 24);
   302→   302→  if (days < 30) {
   303→   303→    return `${days} ${days === 1 ? "day" : "days"} ago`;
   304→   304→  }
   305→   305→  return new Date(ms).toLocaleDateString("en-GB", {
   306→   306→    day: "numeric",
   307→   307→    month: "short",
   308→   308→    year: "numeric",
   309→   309→  });
   310→   310→}

   377→   377→            className={cn(

   436→   436→  return (
   437→   437→    <article
   438→   438→      className={cn(
   439→   439→        "rounded-lg border border-border bg-background p-3 sm:p-4",
   440→   440→        pending && "opacity-60"
   441→   441→      )}

   446→   446→          {entry.authorName || t.anonymousAuthor}

   449→   449→        <time

   451→   451→          dateTime={entry.createdAt}
   452→   452→          title={new Date(entry.createdAt).toLocaleString(
   453→   453→            lang === "en" ? "en-GB" : "sl-SI"
   454→   454→          )}
   455→   455→        >
   456→   456→          {timeAgo(entry.createdAt, tripCreatedAt, lang)}
   457→   457→        </time>
   458→   458→        {entry.rating !== null && entry.rating > 0 && (
   459→   459→          <Stars value={entry.rating} />
   460→   460→        )}
   461→   461→      </div>
   462→   462→
   463→   463→      {/* Kontekst: dan + kraj */}
   464→   464→      {(dayLabel || entry.placeName) && (
   465→   465→        <div className="mb-2 flex flex-wrap items-center gap-1.5 text-xs text-muted-foreground">
   466→   466→          {dayLabel && (
   467→   467→            <span className="inline-flex items-center gap-1 rounded-full bg-muted px-2 py-0.5 font-medium">
   468→   468→              {dayLabel}
   469→   469→            </span>
   470→   470→          )}
   471→   471→          {entry.placeName && (
   472→   472→            <span className="inline-flex items-center gap-1 rounded-full bg-muted px-2 py-0.5">
   473→   473→              <MapPin className="size-3" aria-hidden="true" />
   474→   474→              {entry.placeName}
   475→   475→            </span>
   476→   476→          )}
   477→   477→        </div>
   478→   478→      )}
   479→   479→
   480→   480→      {/* Besedilo (ali urejanje) */}
   481→   481→      {editing ? (
   482→   482→        <div className="space-y-2">
   483→   483→          <Textarea
   484→   484→            value={editText}

   490→   490→          />
   491→   491→          <div className="flex items-center justify-between">
   492→   492→            <span className="text-xs text-muted-foreground">
   493→   493→              {editText.length}/{TEXT_MAX}
   494→   494→            </span>

   496→   496→              <Button
   497→   497→                type="button"
   498→   498→                variant="outline"
   499→   499→                size="sm"

   519→   519→        </div>
   520→   520→      ) : (

   524→   524→      )}
   525→   525→
   526→   526→      {/* Avtorske kontrole — samo lastnik vpisa */}
   527→   527→      {entry.isAuthor && !editing && (
   528→   528→        <div className="mt-2 flex gap-2 print:hidden">
   529→   529→          <Button
   530→   530→            type="button"
   531→   531→            variant="ghost"
   532→   532→            size="sm"
   533→   533→            onClick={startEdit}
   534→   534→            disabled={pending}
   535→   535→            className="h-7 gap-1 px-2 text-xs text-muted-foreground hover:text-foreground"
   536→   536→          >
   537→   537→            <Pencil className="size-3" aria-hidden="true" /> {t.edit}
   538→   538→          </Button>
   539→   539→          <Button
   540→   540→            type="button"
   541→   541→            variant="ghost"
   542→   542→            size="sm"
   543→   543→            onClick={() => void onDelete(entry.id)}
   544→   544→            disabled={pending}
   545→   545→            className="h-7 gap-1 px-2 text-xs text-muted-foreground hover:text-destructive"
   546→   546→          >
   547→   547→            <Trash2 className="size-3" aria-hidden="true" /> {t.delete}

   549→   549→        </div>

   551→   551→
   552→   552→      {/* Urejeno-opomba (če se je besedilo spreminjalo) */}
   553→   553→      {entry.updatedAt !== entry.createdAt && !editing && (

   555→   555→          {t.edited} {timeAgo(entry.updatedAt, tripCreatedAt, lang)}
   556→   556→        </p>
   557→   557→      )}
   558→   558→    </article>
   559→   559→  );
   560→   560→}

   574→   574→  const locale = useLocale();
   575→   575→  const lang: "sl" | "en" = locale === "en" ? "en" : "sl";

   594→   594→
   595→   595→  /** Znova pridobi vpise z isAuthor (GET /api/diary?clientId=). */

   600→   600→          `/api/diary?shareId=${encodeURIComponent(shareId)}&clientId=${encodeURIComponent(cid)}`,
   601→   601→          { cache: "no-store" }
   602→   602→        );
   603→   603→        if (!res.ok) return;

   605→   605→          entries?: DiaryEntry[];

   607→   607→        if (Array.isArray(data?.entries)) {
   608→   608→          setEntries(data.entries);
   609→   609→        }
   610→   610→      } catch {
   611→   611→        // Off-line / napaka — ohranimo trenutno stanje

   624→   624→      const cid = getVoterId();
   625→   625→      if (cid) {

   630→   630→        if (savedName) setAuthorName(savedName);

   635→   635→    } catch {

   639→   639→

   653→   653→        toast({
   654→   654→          title: t.errTextTitle,

   656→   656→          variant: "destructive",
   657→   657→        });

   659→   659→      }

   664→   664→          variant: "destructive",
   665→   665→        });

   725→   725→

   742→   742→      } catch (err) {

   749→   749→        });

   751→   751→        setCreating(false);
   752→   752→      }
   753→   753→    },
   754→   754→    [
   755→   755→      creating,
   756→   756→      shareId,
   757→   757→      clientId,
   758→   758→      text,
   759→   759→      place,
   760→   760→      authorName,
   761→   761→      rating,
   762→   762→      dayValue,

   772→   772→    async (entryId: string, newText: string) => {
   773→   773→      if (!clientId || entryId.startsWith("tmp-")) return;
   774→   774→      const txt = newText.trim();
   775→   775→      if (txt.length < TEXT_MIN) {
   776→   776→        toast({
   777→   777→          title: t.errShortTitle,
   778→   778→          description: t.errShortDesc(),
   779→   779→          variant: "destructive",
   780→   780→        });
   781→   781→        return;
   782→   782→      }
   783→   783→
   784→   784→      setActionPending(entryId);
   785→   785→      const before = entries.find((e) => e.id === entryId)?.text;
   786→   786→      // Optimistična sprememba

   788→   788→        all.map((e) => (e.id === entryId ? { ...e, text: txt } : e))

   862→   862→  // ========================================================================

   865→   865→  const grouped = useMemo(() => {

   868→   868→      label: string;
   869→   869→      items: DiaryEntry[];
   870→   870→    }[] = [];

   886→   886→      else general.items.push(e);

   890→   890→    const dayGroups = groups.filter((g) => g.items.length > 0);
   891→   891→    const generalGroup =
   892→   892→      general.items.length > 0 ? [general] : [];
   893→   893→
   894→   894→    return [...dayGroups, ...generalGroup];
   895→   895→  }, [entries, dayLabels, t]);
   896→   896→
   897→   897→  // Dnevnik se NATISNE (to je naš "photobook") — skriti so samo obrazec,
   898→   898→  // kontrole in interaktivne zvezdice (spodaj s print:hidden).
   899→   899→  return (
   900→   900→    <Card className="overflow-hidden">
   901→   901→      <CardContent className="p-4 sm:p-6">
   902→   902→        <div className="mb-4 flex flex-wrap items-center justify-between gap-3">
   903→   903→          <div className="flex items-center gap-2">
   904→   904→            <BookOpen className="size-5 text-primary" aria-hidden="true" />
   905→   905→            <h2 className="text-lg font-semibold">{t.heading}</h2>
   906→   906→            <span className="text-sm text-muted-foreground">
   907→   907→              {entriesLabel(entries.length, lang)}
   908→   908→            </span>
   909→   909→          </div>
   910→   910→          <Button
   911→   911→            type="button"
   912→   912→            variant={formOpen ? "outline" : "default"}
   913→   913→            size="sm"
   914→   914→            onClick={() => setFormOpen((o) => !o)}
