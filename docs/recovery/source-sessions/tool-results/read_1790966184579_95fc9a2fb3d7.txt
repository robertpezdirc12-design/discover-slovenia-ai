   160→/** "HH:MM-HH:MM" → minute od polnoči; null pri drugačnem formatu (brez ugibanj). */
   161→function parseSlot(slot: string): { startMin: number; endMin: number } | null {
   162→  if (typeof slot !== "string") return null;
   163→  const m = slot.match(/(\d{1,2}):(\d{2})\s*[-–]\s*(\d{1,2}):(\d{2})/);
   164→  if (!m) return null;
   165→  const s = Number(m[1]) * 60 + Number(m[2]);
   166→  const e = Number(m[3]) * 60 + Number(m[4]);
   167→  if (e <= s) return null; // čez noč ali smešno — preskoči
   168→  return { startMin: s, endMin: e };
   169→}
   170→
   171→/** TASK 48 (§4): razvrsti time_slot v tri stanja — parseSlot null je
   172→ *  dvomensen (neformatiran VERSUS obrnjen), invarianta "start < end"
   173→ *  pa zahteva, da OBRNJEN termin sproži napako, ne tiho preskočenje. */
   174→type SlotKind =
   175→  | { kind: "ok"; startMin: number; endMin: number }
   176→  | { kind: "reversed" } // vzorcu ustreza, a konec ≤ začetek (npr. 13:00-09:00)
   177→  | { kind: "unknown" }; // neformatiran — ne trdimo ničesar (§8)
   178→
   179→function classifySlot(slot: unknown): SlotKind {
   180→  if (typeof slot !== "string") return { kind: "unknown" };
   181→  const m = slot.match(/(\d{1,2}):(\d{2})\s*[-–]\s*(\d{1,2}):(\d{2})/);
   182→  if (!m) return { kind: "unknown" };
   183→  const s = Number(m[1]) * 60 + Number(m[2]);
   184→  const e = Number(m[3]) * 60 + Number(m[4]);
   185→  if (e <= s) return { kind: "reversed" };
   186→  return { kind: "ok", startMin: s, endMin: e };
   187→}

   189→/** TASK 48 (§4/§6): koordinate postanka — T1 dataset ALI lastne (supply/chat
   190→ *  postanki imajo lastne lat/lng; enaka semantika kot pravilo missing_coords).
   191→ *  TOČNO (0,0) je null island — geo sentinel "ni podatka", ki ga AI odmev
   192→ *  izpljune namesto koordinat (živ primer: refine "socca" @ 0,0 bi izračunal
   193→ *  ~6.920 km nogo). Enak izid kot manjkajoče koordinate: iskreno
   194→ *  "ne morem preveriti", ne absurdne razdalje.
   195→ *
   196→ *  W11-B (izvožena): to je ZDAJ ena izvožena resnica za resolucijo koordinat
   197→ *  postanka — uporabljajo jo geo-validation, gmaps-day-export (W11-A) in
   198→ *  day-quality (W11-B). Prej tri identične privatne kopije. */
   199→export function coordsOfStop(s: LocationVisit): { lat: number; lng: number } | null {
   200→  const t1 = COORDS.get(s.destination_id);
   201→  if (t1) return t1;
   202→  if (
   203→    typeof s.lat === "number" &&
   204→    typeof s.lng === "number" &&
   205→    Number.isFinite(s.lat) &&
   206→    Number.isFinite(s.lng) &&
   207→    !(s.lat === 0 && s.lng === 0)
   208→  ) {
   209→    return { lat: s.lat, lng: s.lng };
   210→  }
   211→  return null;
   212→}

   424→export function validateItineraryGeo(
   425→  itinerary: Itinerary,
   426→  lang: Lang = "sl",
   427→  /** F5.6: indeks nog (realne ceste, OSRM) — opcijsko; brez njega hevristika. */
   428→  legs?: LegRouteIndex
   429→): GeoValidation {
   430→  const days = Array.isArray(itinerary.days) ? itinerary.days : [];
   431→  const dayMetrics: DayGeoMetrics[] = [];
   432→  const issues: GeoValidationIssue[] = [];
   433→  // F5.6: metoda razdalj (razkritje v odgovoru) — štejemo porabljene noge.
   434→  const usedLegs: LegRouteIndex = new Map();
   435→
   436→  // F5.5: datum tega dneva ( dan N = tripStartDate + N-1) — SAMO z znanim
   437→  // datumom odhoda. Brez datuma pravili odpiralnih časov NE veljata
   438→  // ( neznani datum = ni poštene trditve o dnevu v tednu/mesecu).
   439→  const tripStart =
   440→    typeof itinerary.tripStartDate === "string"
   441→      ? itinerary.tripStartDate
   442→      : null;
   443→
   444→  for (const day of days) {
   445→    const stops = Array.isArray(day.locations) ? day.locations : [];
   446→    const dayNo = typeof day.day === "number" ? day.day : 0;
   447→
   448→    // Datum dneva za pravili odpiralnih časov ( izven zanke over stops —
   449→    // izračun enkrat na dan)
   450→    const dayDateMs = tripStart
   451→      ? (() => {
   452→          const iso = dayISOForDayNumber(tripStart, dayNo);
   453→          return iso ? parseISODateLocal(iso) : null;
   454→        })()
   455→      : null;
   456→    const dayMonth = dayDateMs !== null ? new Date(dayDateMs).getMonth() + 1 : null;
   457→    const dayWeekday = dayDateMs !== null ? new Date(dayDateMs).getDay() : null;
   458→
   459→    // --- pravilo 7: manjkajoče koordinate (neznan ID → ne moremo računati) ---
   460→    // AUDIT 42 (42-d RED #2): supply/chat postanki imajo LASTNE lat/lng —
   461→    // prej je vsak ne-T1 id (tudi z veljavnimi koordinatami) sprožil
   462→    // missing_coords NAPAKO. Zdaj: neznanim ID-jem z lastnimi končnimi
   463→    // koordinatami napaka NE pade (enaka semantika kot store.ts coordsOf).
   464→    for (const s of stops) {
   465→      // TASK 48: (0,0) null island šteje kot MANJKAJOČE koordinate —
   466→      // enaka odločitev kot coordsOfStop za noge (iskreno "ne morem
   467→      // preveriti", ne absurdne razdalje).
   468→      const hasOwnCoords = coordsOfStop(s) !== null;
   469→      if (!COORDS.has(s.destination_id) && !hasOwnCoords) {
   470→        issues.push({
   471→          day: dayNo,
   472→          level: "error",
   473→          rule: "missing_coords",
   474→          message: msgMissingCoords(
   475→            lang,
   476→            s.destination_name ?? s.destination_id,
   477→            s.destination_id
   478→          ),
   479→        });
   480→      }
   481→    }
   482→
   483→    // --- TASK 48 (§4 časovne invariante): termin in trajanje postanka ---
   484→    // OBRNJEN termin (start ≥ end) se prej TIHO preskočil (parseSlot → null
   485→    // brez issue); zdaj sproži napako. Neformatiran termin ostaja brez
   486→    // trditve (unknown is unknown). duration ≤ 0 je fizično nemogoč
   487→    // (sanitizacija ga sicer clampa na novih načrtih — to je varovalka za
   488→    // stare/shranjene načrte, ki sanitizacije še niso šli).
   489→    for (const s of stops) {
   490→      const slotKind = classifySlot(s.time_slot);
   491→      if (slotKind.kind === "reversed") {
   492→        issues.push({
   493→          day: dayNo,
   494→          level: "error",
   495→          rule: "time_slot_invalid",
   496→          message: msgTimeSlotInvalid(
   497→            lang,
   498→            s.destination_name ?? s.destination_id,
   499→            s.time_slot
   500→          ),
   501→        });
   502→      }
   503→      if (typeof s.duration !== "number" || !Number.isFinite(s.duration) || s.duration <= 0) {
   504→        issues.push({
   505→          day: dayNo,
   506→          level: "error",
   507→          rule: "duration_invalid",
   508→          message: msgDurationInvalid(
   509→            lang,
   510→            s.destination_name ?? s.destination_id
   511→          ),
   512→        });
   513→      }
   514→    }
   515→
   516→    // --- pravili 9 + 10 ( F5.5): odpiralni časi — zaprtje meseca/dneva v
   517→    //     tednu. SAMO z znanim datumom + SAMO za preverjene vnose. Raven
   518→    //     zaprtja: destination → ERROR ( soteska JE kraj), mainAttraction →
   519→    //     WARN ( mesto odprto, grad/muzej zaprt). Vir je VEDNO v sporočilu. ---
   520→    if (dayMonth !== null) {
   521→      for (const s of stops) {
   522→        const opening = OPENING.get(s.destination_id);
   523→        const dest = DEST_BY_ID.get(s.destination_id);
   524→        if (!opening || !dest) continue;
   525→        const name = s.destination_name ?? dest.name;
   526→
   527→        if (
   528→          opening.closedMonths &&
   529→          opening.closedMonths.includes(dayMonth)
   530→        ) {
   531→          issues.push({
   532→            day: dayNo,
   533→            level: opening.closureLevel === "destination" ? "error" : "warn",
   534→            rule: "closed_month",
   535→            message: msgClosedMonth(
   536→              lang,
   537→              name,
   538→              lang === "en" ? opening.noteEn : opening.note,
   539→              opening.source,
   540→              opening.closureLevel === "destination"
   541→            ),
   542→          });
   543→        } else if (
   544→          dayWeekday !== null &&
   545→          opening.closedWeekdays &&
   546→          opening.closedWeekdays.includes(dayWeekday)
   547→        ) {
   548→          issues.push({
   549→            day: dayNo,
   550→            level: "warn",
   551→            rule: "closed_weekday",
   552→            message: msgClosedWeekday(
   553→              lang,
   554→              name,
   555→              dayWeekday,
   556→              lang === "en" ? opening.noteEn : opening.note,
   557→              opening.source
   558→            ),
   559→          });
   560→        }
   561→      }
   562→    }
   563→
   564→    // --- zaporedne noge: razdalje, vožnja, urnik ---
   565→    // TASK 48 (§4/§6): koordinate se razrešijo iz T1 dataseta ALI lastnih
   566→    // lat/lng postanka (supply/chat kraji) — prej je bila KATERAKOLI noga s
   567→    // ne-T1 koncem IZVZETA iz vseh treh preverjanj (razdalja, vožnja, urnik),
   568→    // zdaj supply postanki sodelujejo enako kot T1 (hevristika haversine ×
   569→    // ROAD_FACTOR pošteno razkrita z "~", enako kot ob padcu OSRM).
   570→    // Urnik (prekrivanje) se preverja NEODVISNO od koordinat — za prekrivanje
   571→    // terminov fizična lokacija ni potrebna, samo oba parsable termina.
   572→    let kmStraight = 0; // hevristična razdalja (rezerva, kadar ni indeksa)
   573→    let driveH = 0;
   574→    let dayKm = 0; // F5.6: seštevek km nog (realne ceste, kadar so na voljo)
   575→    for (let i = 0; i < stops.length - 1; i++) {
   576→      const a = coordsOfStop(stops[i]);
   577→      const b = coordsOfStop(stops[i + 1]);
   578→      let legH: number | null = null;
   579→      if (a && b) {
   580→        const leg = legs?.get(
   581→          legKey(stops[i].destination_id, stops[i + 1].destination_id)
   582→        );
   583→        const straight = haversineKm(a, b);
   584→        const roadKm = leg ? leg.km : round5(straight * ROAD_FACTOR);
   585→        legH = leg ? leg.min / 60 : (straight * ROAD_FACTOR) / AVG_SPEED_KMH;
   586→        kmStraight += straight;
   587→        driveH += legH;
   588→        dayKm += roadKm;
   589→        if (leg) {
   590→          usedLegs.set(
   591→            legKey(stops[i].destination_id, stops[i + 1].destination_id),
   592→            leg
   593→          );
   594→        }
   595→
   596→        // pravilo 3: zaporedna razdalja
   597→        if (roadKm > THRESHOLDS.legKm.error) {
   598→          issues.push({
   599→            day: dayNo,
   600→            level: "error",
   601→            rule: "leg_distance",
   602→            message: msgLegDistance(
   603→              lang,
   604→              stops[i].destination_name ?? stops[i].destination_id,
   605→              stops[i + 1].destination_name ?? stops[i + 1].destination_id,
   606→              roadKm,
   607→              "error"
   608→            ),
   609→          });
   610→        } else if (roadKm > THRESHOLDS.legKm.warn) {
   611→          issues.push({
   612→            day: dayNo,
   613→            level: "warn",
   614→            rule: "leg_distance",
   615→            message: msgLegDistance(
   616→              lang,
   617→              stops[i].destination_name ?? stops[i].destination_id,
   618→              stops[i + 1].destination_name ?? stops[i + 1].destination_id,
   619→              roadKm,
   620→              "warn"
   621→            ),
   622→          });
   623→        }
   624→      }
   625→
   626→      // pravilo 5: časovna združljivost (SAMO če sta termina parsable —
   627→      // drugače preskoči, brez izmišljenih časov). Prekrivanje se preverja
   628→      // tudi BREZ koordinat (§4: previous.end ≤ next.start, ko sta termina
   629→      // časovno fiksna); primerjava vrzeli z vožnjo le ob znani nogi (§6).
   630→      const sa = parseSlot(stops[i].time_slot);
   631→      const sb = parseSlot(stops[i + 1].time_slot);
   632→      if (sa && sb) {
   633→        const gapH = (sb.startMin - sa.endMin) / 60;
   634→        if (gapH < 0) {
   635→          issues.push({
   636→            day: dayNo,
   637→            level: "error",
   638→            rule: "schedule_overlap",
   639→            message: msgScheduleOverlap(
   640→              lang,
   641→              stops[i].time_slot,
   642→              stops[i + 1].time_slot
   643→            ),
   644→          });
   645→        } else if (legH !== null && gapH + THRESHOLDS.scheduleTightBuffer < legH) {
   646→          issues.push({
   647→            day: dayNo,
   648→            level: "error",
   649→            rule: "schedule_gap",
   650→            message: msgScheduleGap(
   651→              lang,
   652→              stops[i].destination_name ?? stops[i].destination_id,
   653→              stops[i + 1].destination_name ?? stops[i + 1].destination_id,
   654→              gapH,
   655→              legH,
   656→              "error"
   657→            ),
   658→          });
   659→        } else if (legH !== null && gapH < legH) {
   660→          issues.push({
   661→            day: dayNo,
   662→            level: "warn",
   663→            rule: "schedule_gap",
   664→            message: msgScheduleGap(
   665→              lang,
   666→              stops[i].destination_name ?? stops[i].destination_id,
   667→              stops[i + 1].destination_name ?? stops[i + 1].destination_id,
   668→              gapH,
   669→              legH,
   670→              "warn"
   671→            ),
   672→          });
   673→        }
   674→      }
   675→    }
   676→
   677→    const roadKmDay = dayKm > 0 ? round5(dayKm) : round5(kmStraight * ROAD_FACTOR);
   678→    const drivingMinutes = round5(driveH * 60);
   679→    const activityMinutes = round5(
   680→      stops.reduce((s, x) => s + (Number.isFinite(x.duration) ? x.duration : 0), 0) * 60
   681→    );
   682→    const loadMinutes = drivingMinutes + activityMinutes;
   683→
   684→    // pravilo 1: kilometri na dan
   685→    if (roadKmDay > THRESHOLDS.dayKm.error) {
   686→      issues.push({
   687→        day: dayNo,
   688→        level: "error",
   689→        rule: "day_km",
   690→        message: msgDayKm(lang, roadKmDay),
   691→      });
   692→    } else if (roadKmDay > THRESHOLDS.dayKm.warn) {
   693→      issues.push({
   694→        day: dayNo,
   695→        level: "warn",
   696→        rule: "day_km",
   697→        message: msgDayKm(lang, roadKmDay),
   698→      });
   699→    }
   700→
   701→    // pravilo 2: število postankov
   702→    if (stops.length > THRESHOLDS.dayStops.error) {
   703→      issues.push({
   704→        day: dayNo,
   705→        level: "error",
   706→        rule: "day_stops",
   707→        message: msgDayStops(lang, stops.length),
   708→      });
   709→    } else if (stops.length > THRESHOLDS.dayStops.warn) {
   710→      issues.push({
   711→        day: dayNo,
   712→        level: "warn",
   713→        rule: "day_stops",
   714→        message: msgDayStops(lang, stops.length),
   715→      });
   716→    }
   717→
   718→    // pravilo 4: obseg dneva (aktivnosti + vožnje)
   719→    const loadH = loadMinutes / 60;
   720→    if (loadH > THRESHOLDS.dayLoadHours.error) {
   721→      issues.push({
   722→        day: dayNo,
   723→        level: "error",
   724→        rule: "day_overload",
   725→        message: msgDayOverload(
   726→          lang,
   727→          Math.round((activityMinutes / 60) * 10) / 10,
   728→          Math.round((drivingMinutes / 60) * 10) / 10
   729→        ),
   730→      });
   731→    } else if (loadH > THRESHOLDS.dayLoadHours.warn) {
   732→      issues.push({
   733→        day: dayNo,
   734→        level: "warn",
   735→        rule: "day_overload",
   736→        message: msgDayOverload(
   737→          lang,
   738→          Math.round((activityMinutes / 60) * 10) / 10,
   739→          Math.round((drivingMinutes / 60) * 10) / 10
   740→        ),
   741→      });
   742→    }
   743→
   744→    // pravilo 6: isti kraj dvakrat isti dan
   745→    const seen = new Set<string>();
   746→    for (const s of stops) {
   747→      if (seen.has(s.destination_id)) {
   748→        issues.push({
   749→          day: dayNo,
   750→          level: "warn",
   751→          rule: "duplicate_stop",
   752→          message: msgDuplicateStop(lang, s.destination_name ?? s.destination_id),
   753→        });
   754→      }
   755→      seen.add(s.destination_id);
   756→    }
   757→
   758→    dayMetrics.push({
   759→      day: dayNo,
   760→      stops: stops.length,
   761→      km: roadKmDay,
   762→      drivingMinutes,
   763→      activityMinutes,
   764→      loadMinutes,
   765→    });
   766→  }
   767→
   768→  const tripKm = dayMetrics.reduce((s, d) => s + d.km, 0);
   769→  const worst: GeoValidation["worst"] = issues.some((i) => i.level === "error")
   770→    ? "error"
   771→    : issues.some((i) => i.level === "warn")
   772→      ? "warn"
   773→      : "ok";
   774→
   775→  // F5.6: metoda razdalj tega izračuna (razkritje v UI). Brez indeksa
   776→  // (client, stari načrti) polje manjka → panel izpiše hevristiko.
   777→  let method: RoutingMethod | undefined;
   778→  if (legs && usedLegs.size > 0) method = legIndexMethod(usedLegs);
   779→  else if (legs) method = "heuristic";
   780→
   781→  // TASK 4 / K-3 (UX FIX PASS): ali se je plast odpiralnih časov DEJANSKO
   782→  // izvedla — zahteva znan datum odhoda (tripStart). Brez datuma pravili
   783→  // closed_month/closed_weekday NE tečeta (`if (dayMonth !== null)`), zato
   784→  // je "✓ Odprto ob tvojem času" nad takim načrtom LAŽNA trditev. Flag je
   785→  // izhodišče za PlannerTrustLine (NE izriše ✓, če plasti ni bilo).
   786→  const openingHoursChecked = tripStart !== null && days.length > 0;
   787→
   788→  return {
   789→    days: dayMetrics,
   790→    issues,
   791→    tripKm,
   792→    worst,
   793→    openingHoursChecked,
   794→    ...(method ? { method } : {}),
   795→  };
   796→}
