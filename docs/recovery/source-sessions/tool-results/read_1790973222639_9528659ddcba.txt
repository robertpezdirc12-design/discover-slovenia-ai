   220→function slUnit(
   221→  n: number,
   222→  one: string,
   223→  two: string,
   224→  few: string,
   225→  many: string
   226→): string {
   227→  if (n === 1) return one;
   228→  const r = n % 100;
   229→  if (r === 2) return two;
   230→  if (r === 3 || r === 4) return few;
   231→  return many;
   232→}

   234→/** "pred 2 min", "pred 3 h" — relativni čas z varnostnim stropom. */
   235→function slTimeAgo(iso: string, floorIso?: string): string {
   236→  const ms = Date.parse(iso);
   237→  if (Number.isNaN(ms)) return "neznano";
   238→
   239→  // Sanity cap: vpis ne more biti starejši od potovanja
   240→  if (floorIso) {
   241→    const floor = Date.parse(floorIso);
   242→    if (!Number.isNaN(floor) && ms < floor) {
   243→      return new Date(ms).toLocaleDateString("sl-SI", {
   244→        day: "numeric",
   245→        month: "short",
   246→        year: "numeric",
   247→      });
   248→    }
   249→  }
   250→
   251→  const seconds = Math.max(0, Math.floor((Date.now() - ms) / 1000));
   252→  if (seconds < 45) return "pravkar";
   253→
   254→  const minutes = Math.floor(seconds / 60);
   255→  if (minutes < 60) {
   256→    return `pred ${minutes} ${slUnit(minutes, "minuto", "minutama", "minutami", "minutami")}`;
   257→  }
   258→  const hours = Math.floor(minutes / 60);
   259→  if (hours < 24) {
   260→    return `pred ${hours} ${slUnit(hours, "uro", "urama", "urami", "urami")}`;
   261→  }
   262→  const days = Math.floor(hours / 24);
   263→  if (days < 30) {
   264→    return `pred ${days} ${slUnit(days, "dnem", "dnevoma", "dnevi", "dnevi")}`;
   265→  }
   266→  return new Date(ms).toLocaleDateString("sl-SI", {
   267→    day: "numeric",
   268→    month: "short",
   269→    year: "numeric",
   270→  });
   271→}
   272→
   273→/** EN dvojnik slTimeAgo ("2 min ago") — isti varnostni strop, "en-GB" datumi. */
   274→function enTimeAgo(iso: string, floorIso?: string): string {
   275→  const ms = Date.parse(iso);
   276→  if (Number.isNaN(ms)) return "unknown";
   277→
   278→  // Sanity cap: vpis ne more biti starejši od potovanja
   279→  if (floorIso) {
   280→    const floor = Date.parse(floorIso);
   281→    if (!Number.isNaN(floor) && ms < floor) {
   282→      return new Date(ms).toLocaleDateString("en-GB", {
   283→        day: "numeric",
   284→        month: "short",
   285→        year: "numeric",
   286→      });
   287→    }
   288→  }
   289→
   290→  const seconds = Math.max(0, Math.floor((Date.now() - ms) / 1000));
   291→  if (seconds < 45) return "just now";
   292→
   293→  const minutes = Math.floor(seconds / 60);
   294→  if (minutes < 60) {
   295→    return `${minutes} min ago`;
   296→  }
   297→  const hours = Math.floor(minutes / 60);
   298→  if (hours < 24) {
   299→    return `${hours} h ago`;
   300→  }
   301→  const days = Math.floor(hours / 24);
   302→  if (days < 30) {
   303→    return `${days} ${days === 1 ? "day" : "days"} ago`;
   304→  }
   305→  return new Date(ms).toLocaleDateString("en-GB", {
   306→    day: "numeric",
   307→    month: "short",
   308→    year: "numeric",
   309→  });
   310→}

   312→/** ISSUE #24 Sklop 1 (1.164.0): jezikovni dispenser — SL pot ostane
   313→ *  byte-identična (slTimeAgo), EN gre po enTimeAgo. */
   314→function timeAgo(
   315→  iso: string,
   316→  floorIso: string | undefined,
   317→  lang: "sl" | "en"
   318→): string {
   319→  return lang === "en" ? enTimeAgo(iso, floorIso) : slTimeAgo(iso, floorIso);
   320→}

   342→function Stars({ value, onChange, size = "sm", ariaLabel }: StarsProps) {
   343→  // ISSUE #24 Sklop 1 (1.164.0): privzete aria oznake zvezdic po locale gledalca.
   344→  const locale = useLocale();

   348→  const cls = size === "md" ? "size-5" : "size-4";
   349→  return (
   350→    <div
   351→      className={cn(
   352→        "flex items-center gap-0.5",
   353→        interactive && "print:hidden"
   354→      )}
   355→      role={interactive ? "radiogroup" : "img"}
   356→      aria-label={
   357→        ariaLabel ?? (value > 0 ? t.starsLabel(value) : t.noRating)
   358→      }
   359→    >
   360→      {[1, 2, 3, 4, 5].map((i) => (
   361→        <button
   362→          key={i}
   363→          type="button"
   364→          disabled={!interactive}
   365→          onClick={interactive ? () => onChange(value === i ? 0 : i) : undefined}
   366→          className={cn(
   367→            "rounded transition-colors",
   368→            interactive &&
   369→              "cursor-pointer hover:scale-110 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",
   370→            !interactive && "cursor-default"
   371→          )}
   372→          aria-label={interactive ? t.rateLabel(i) : undefined}
   373→          aria-checked={interactive ? value === i : undefined}
   374→          role={interactive ? "radio" : undefined}
   375→        >
   376→          <Star
   377→            className={cn(
   378→              cls,
   379→              i <= value
   380→                ? "fill-amber-500 text-amber-500"
   381→                : "text-muted-foreground/40"
   382→            )}
   383→            aria-hidden="true"
   384→          />
   385→        </button>
   386→      ))}
   387→    </div>
   388→  );
   389→}

   529→          <Button
   530→            type="button"
   531→            variant="ghost"
   532→            size="sm"
   533→            onClick={startEdit}
   534→            disabled={pending}
   535→            className="h-7 gap-1 px-2 text-xs text-muted-foreground hover:text-foreground"
   536→          >
   537→            <Pencil className="size-3" aria-hidden="true" /> {t.edit}
   538→          </Button>
   539→          <Button
   540→            type="button"
   541→            variant="ghost"
   542→            size="sm"
   543→            onClick={() => void onDelete(entry.id)}
   544→            disabled={pending}
   545→            className="h-7 gap-1 px-2 text-xs text-muted-foreground hover:text-destructive"
   546→          >
   547→            <Trash2 className="size-3" aria-hidden="true" /> {t.delete}
   548→          </Button>
   549→        </div>
   550→      )}
   551→
   552→      {/* Urejeno-opomba (če se je besedilo spreminjalo) */}
   553→      {entry.updatedAt !== entry.createdAt && !editing && (
   554→        <p className="mt-1.5 text-xs text-muted-foreground">
   555→          {t.edited} {timeAgo(entry.updatedAt, tripCreatedAt, lang)}
   556→        </p>
   557→      )}
   558→    </article>
   559→  );
   560→}
