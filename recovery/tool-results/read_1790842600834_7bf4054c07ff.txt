  1195→### 12.3 Implementacijske specifikacije
  1196→
  1197→```typescript
  1198→// src/lib/auth-guards.ts
  1199→type Role = "visitor" | "user" | "provider" | "premium" | "enterprise" | "moderator" | "admin" | "super_admin";
  1200→
  1201→interface Permission {
  1202→  resource: "listing" | "product" | "experience" | "sponsorship" | "analytics" | "admin";
  1203→  action: "read" | "create" | "update" | "delete" | "approve" | "manage";
  1204→  scope: "own" | "all";
  1205→}
  1206→
  1207→// Role → Permissions mapping (konstanta)
  1208→const ROLE_PERMISSIONS: Record<Role, Permission[]> = {
  1209→  visitor: [
  1210→    { resource: "listing", action: "read", scope: "all" },
  1211→    // ... samo read na public resource
  1212→  ],
  1213→  provider: [
  1214→    { resource: "listing", action: "create", scope: "own" },
  1215→    { resource: "listing", action: "update", scope: "own" },
  1216→    { resource: "listing", action: "read", scope: "all" },
  1217→  ],
  1218→  admin: [
  1219→    { resource: "listing", action: "*", scope: "all" },
  1220→    { resource: "sponsorship", action: "*", scope: "all" },
  1221→    // ... full access
  1222→  ],
  1223→};
  1224→
  1225→function canPerform(role: Role, perm: Permission): boolean {
  1226→  return ROLE_PERMISSIONS[role].some(p =>
  1227→    p.resource === perm.resource &&
  1228→    (p.action === perm.action || p.action === "*") &&
  1229→    p.scope === perm.scope
  1230→  );
  1231→}
  1232→```
