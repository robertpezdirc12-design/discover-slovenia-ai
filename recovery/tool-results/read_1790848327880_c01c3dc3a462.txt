   147→// ---------------------------------------------------------------------------
   148→// ENV DOSTOP (§6) — Boolean bralci PO IMENU (nikoli vrednosti ven)
   149→// ---------------------------------------------------------------------------
   150→// Imena so IZPELJANA iz registra (envKeys) — en sam vir resnice. Bralci
   151→// so namerno ločeni od affiliate.ts: tam se vrednost UPORABI za URL, tu
   152→// se zanjo izrazi SAMO prisotnost (matrika/admin/dokumentacija).
   153→
   154→function envPresent(name: string | undefined): EnvAccessCheck | null {
   155→  if (!name) return null;
   156→  const v = process.env[name];
   157→  if (typeof v !== "string") return { envVar: name, present: false };
   158→  // URL spremenljivke: „prisoten“ pomeni VELJAVEN https URL (ista
   159→  // semantika kot affiliate.ts isValidHttpsUrl — http/smét = neveljaven
   160→  // vnos NI konfiguracija). ID/ključ spremenljivke: ne-prazno po trimu.
   161→  if (name.endsWith("_URL")) {
   162→    return { envVar: name, present: isValidHttpsEnvUrl(v.trim()) };
   163→  }
   164→  return { envVar: name, present: v.trim().length > 0 };
   165→}
   166→
   167→/** Lokalna kopija https validacije (brez uvoza affiliate.ts — modul ostane
   168→ *  lahek; vrednost NE zapusti funkcije, ven gre samo Boolean). */
   169→function isValidHttpsEnvUrl(value: string): boolean {
   170→  try {
   171→    const u = new URL(value);
   172→    return u.protocol === "https:" && u.hostname.includes(".");
   173→  } catch {
   174→    return false;
   175→  }
   176→}

   178→/** ENV dostop providerja (AFFILIATE + API ločeno, §6). */
   179→export function providerEnvAccess(slug: ProviderSlug): ProviderEnvAccess {
   180→  const entry = PROVIDER_REGISTRY.find((p) => p.slug === slug);
   181→  if (!entry) {
   182→    return { affiliate: [], api: [], productionConfigured: false };
   183→  }
   184→  const affiliate = (entry.envKeys.affiliate ?? [])
   185→    .map(envPresent)
   186→    .filter((c): c is EnvAccessCheck => c != null);
   187→  const api = (entry.envKeys.api ?? [])
   188→    .map(envPresent)
   189→    .filter((c): c is EnvAccessCheck => c != null);
   190→  // „Ključni" vnos = CREDENTIAL-LIKE spremenljivka (ID/ključ/žeton/URL
   191→  // sledenja). NAMERNO IZKLJUČENI: _BASE (preklop produkcija/sandbox brez
   192→  // ključa ne pomeni konfiguracije), _DIR (pot do lokalnega dataseta) in
   193→  // _ORIGIN (TASK 53: operaterska konfiguracija izhodišča letov — IATA
   194→  // koda brez žetona NI konfiguracija vira).
   195→  // Vir brez obeh skupin (OSM/STO/own) → konfiguriranost oceni MATRIKA
   196→  // (PRODUCTION_ACTIVE pomeni: ta vrsta supply-a DEJANSKO teče — npr. STO
   197→  // kot RAG vir teče, čeprav NI sloj zemljevida; own še ne).
   198→  const needsEnv = affiliate.length > 0 || api.length > 0;
   199→  const credentialLike = (name: string) =>
   200→    !name.endsWith("_BASE") && !name.endsWith("_DIR") && !name.endsWith("_ORIGIN");
   201→  // API poverilnice: VSE credential-like vnosе api skupine morajo biti
   202→  // prisotne (TASK 53 §21: airalo OAuth2 zahteva OBE — CLIENT_ID +
   203→  // CLIENT_SECRET; DELNA konfiguracija = NE konfigurirano, iskreno
   204→  // fail-closed). Eno-ključni viri (viator/tiqets/…) imajo en sam vnos
   205→  // → every() je tam ekvivalenten some().
   206→  const apiCreds = api.filter((c) => credentialLike(c.envVar));
   207→  const apiConfigured = apiCreds.length > 0 && apiCreds.every((c) => c.present);
   208→  // Affiliate poverilnica: ENA zadostuje (monetizacijska plast je
   209→  // samostojen produkcijski dosežek — affiliate CTA teče brez API ključa).
   210→  const affiliateConfigured = affiliate.some(
   211→    (c) => c.present && credentialLike(c.envVar)
   212→  );
   213→  const productionConfigured = needsEnv
   214→    ? apiConfigured || affiliateConfigured
   215→    : MATRIX[entry.slug as ProviderSlug]?.stage === "PRODUCTION_ACTIVE";
   216→  return { affiliate, api, productionConfigured };
   217→}

   487→/** Celotna MASTER matrika (skupaj z env dostopom instance). */
   488→export function productionMatrix(): ProductionMatrixEntry[] {
   489→  return PROVIDER_REGISTRY.map((p) => ({
   490→    slug: p.slug,
   491→    ...MATRIX[p.slug],
   492→  }));
   493→}
   494→
   495→/** Posamezen vnos matrike ( ali undefined). */
   496→export function getProductionMatrixEntry(
   497→  slug: string
   498→): ProductionMatrixEntry | undefined {
   499→  const entry = PROVIDER_REGISTRY.find((p) => p.slug === slug);
   500→  if (!entry) return undefined;
   501→  const m = MATRIX[entry.slug as ProviderSlug];
   502→  return m ? { slug: entry.slug, ...m } : undefined;
   503→}
   504→
   505→/**
   506→ * Stopnje, ki jih je provider DEJANSKO prešel (za prikaz verige §0).
   507→ * Vrne urejen seznam vseh stopenj DO (vključno) trenutne — npr.
   508→ * kiwitaxi → vseh 11; viator → 4 (DISCOVERED..CODE_READY).
   509→ */
   510→export function reachedStages(stage: LifecycleStage): LifecycleStage[] {
   511→  const idx = LIFECYCLE_STAGES.indexOf(stage);
   512→  return LIFECYCLE_STAGES.slice(0, idx + 1);
   513→}
   514→
   515→/**
   516→ * Sklepna ocena „production readiness“ celotnega sistema: števci po
   517→ * stopnji + števci blokirnih razlogov (administrativni povzetek).
   518→ */
   519→export function productionSummary(): {
   520→  total: number;
   521→  byStage: Partial<Record<LifecycleStage, number>>;
   522→  byBlockReason: Partial<Record<BlockReason, number>>;
   523→  productionActive: ProviderSlug[];
   524→} {
   525→  const byStage: Partial<Record<LifecycleStage, number>> = {};
   526→  const byBlockReason: Partial<Record<BlockReason, number>> = {};
   527→  const productionActive: ProviderSlug[] = [];
   528→  for (const e of productionMatrix()) {
   529→    byStage[e.stage] = (byStage[e.stage] ?? 0) + 1;
   530→    if (e.blockedReason) {
   531→      byBlockReason[e.blockedReason] = (byBlockReason[e.blockedReason] ?? 0) + 1;
   532→    }
   533→    if (e.stage === "PRODUCTION_ACTIVE") productionActive.push(e.slug);
   534→  }
   535→  return {
   536→    total: PROVIDER_REGISTRY.length,
   537→    byStage,
   538→    byBlockReason,
   539→    productionActive,
   540→  };
   541→}
