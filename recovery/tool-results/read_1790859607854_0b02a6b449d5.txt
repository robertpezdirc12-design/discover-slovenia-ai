    96→function formatDate(iso: string, lang: "sl" | "en"): string {
    97→  try {
    98→    return new Intl.DateTimeFormat(lang === "en" ? "en-GB" : "sl-SI", {
    99→      day: "numeric",
   100→      month: "long",
   101→      year: "numeric",
   102→    }).format(new Date(iso));
   103→  } catch {
   104→    return iso.slice(0, 10);
   105→  }
   106→}

   234→export function MojaPotovanjaView() {
   235→  const router = useRouter();
   236→  const { toast } = useToast();
   237→  const { data: session, status } = useSession();
   238→  // F4-C: jezik chroma (SL privzet — EN le, kadar locale zahteva; P4-8).
   239→  const locale = useLocale();
   240→  const lang: "sl" | "en" = locale === "en" ? "en" : "sl";
   241→
   242→  const [data, setData] = useState<TripsResponse | null>(null);
   243→  const [loadError, setLoadError] = useState<string | null>(null);
   244→  const [resending, setResending] = useState(false);
   245→  // TASK 4 / K-6: gostova LOKALNA potovanja (hidrirajo se TEKOM mounta —
   246→  // localStorage je klient-only, da ni hydration mismatcha).
   247→  const [localTrips, setLocalTrips] = useState<TrackedTrip[] | null>(null);
   248→
   249→  const isUserSession =
   250→    status === "authenticated" && session?.user?.accountType === "user";
   251→
   252→  // TASK 4 / K-6 (UX FIX PASS): NEPREUSMERJENI gost — prej je klik
   253→  // "Moja potovanja" gosta vrgel na /prijava BREZ sporočila, da je pravkar
   254→  // shranjen načrt DEJANSKO še dostopen (deljiva povezava + localStorage).
   255→  // Prelom zlate poti točno pri MY TRIP (živi dokaz revizije, korak 10).
   256→  // Zdaj: gost vidi SVOJA lokalna potovanja + iskreno ponudbo računa.
   257→  // B2B seja še vedno nima kaj iskati tukaj — nazaj na domačo stran.
   258→  useEffect(() => {
   259→    if (
   260→      status === "authenticated" &&
   261→      session?.user?.accountType !== "user"
   262→    ) {
   263→      router.replace("/");
   264→    }
   265→  }, [status, session, router]);
   266→
   267→  // K-6: hidratacija lokalnih potovanj (samo gost — prijavljeni uporablja
   268→  // strežniški seznam iz /api/user/trips + claim plast pri prijavi).
   269→  useEffect(() => {
   270→    if (isUserSession) return;
   271→    const hydrate = setTimeout(() => setLocalTrips(getSavedTrips()), 0);
   272→    return () => clearTimeout(hydrate);
   273→  }, [isUserSession]);
   274→
   275→  // Nalaganje podatkov (samo za B2C sejo)
   276→  useEffect(() => {
   277→    if (!isUserSession) return;
   278→    let cancelled = false;
   279→
   280→    // TASK 8 / F2-A: pull sinhronizacija zbirke "Moja pot" — prinese
   281→    // predmete z drugih naprav (union-merge v localStorage; MyTripView se
   282→    // osveži prek dogodka dai:my-trip-changed). Neblokirajoče + fail-open.
   283→    void syncMyTripToServer();
   284→
   285→    fetch("/api/user/trips", { cache: "no-store" })
   286→      .then(async (r) => {
   287→        if (!r.ok) {
   288→          const d = await r.json().catch(() => ({}));
   289→          throw new Error(d?.error ?? L.errors.loadFailed[lang]);
   290→        }
   291→        return r.json() as Promise<TripsResponse>;
   292→      })
   293→      .then((d) => {
   294→        if (!cancelled) setData(d);
   295→      })
   296→      .catch((e) => {
   297→        if (!cancelled) {
   298→          setLoadError(
   299→            e instanceof Error ? e.message : L.errors.loadFailed[lang]
   300→          );
   301→        }
   302→      });
   303→
   304→    return () => {
   305→      cancelled = true;
   306→    };
   307→  }, [isUserSession, lang]);
   308→
   309→  // Ponovno pošiljanje verifikacijske povezave
   310→  const resendVerification = async () => {
   311→    setResending(true);
   312→    try {
   313→      const res = await fetch("/api/user/verify-email", {
   314→        method: "POST",
   315→        headers: { "Content-Type": "application/json" },
   316→        body: JSON.stringify({ action: "request" }),
   317→      });
   318→      const d = await res.json().catch(() => ({}));
   319→      if (!res.ok) throw new Error(d?.error ?? L.errors.sendFailed[lang]);
   320→      toast({
   321→        title: L.toast.linkSent[lang],
   322→        description:
   323→          d?.message ??
   324→          L.toast.checkInbox[lang],
   325→      });
   326→    } catch (err) {
   327→      toast({
   328→        variant: "destructive",
   329→        title: L.errorTitle[lang],
   330→        description: err instanceof Error ? err.message : L.toast.tryAgain[lang],
   331→      });
   332→    } finally {
   333→      setResending(false);
   334→    }
   335→  };
   336→
   337→  // Spinner SAMO med nalaganjem seje (gost ne čaka več na preusmeritev —
   338→  // TASK 4 / K-6: takoj dobi svoj pogled). TASK 8 / F3-B: LoadingState
   339→  // (block) — oznaka je L-pattern SL/EN (družinski privzetek).
   340→  if (status === "loading" || (status === "authenticated" && !isUserSession)) {
   341→    return (
   342→      <main
   343→        className="flex-1 flex items-center justify-center py-24"
   344→        aria-busy="true"
   345→      >
   346→        <LoadingState variant="block" />
   347→      </main>
   348→    );
   349→
   350→  }
   351→
   352→  // TASK 4 / K-6: GOSTOV POGLED — lokalna potovanja te naprave + iskrena
   353→  // ponudba računa (sinhronizacija). Ni login zida: načrti so DEJANSKO
   354→  // dostopni (deljive povezave so javne; localStorage nosi seznam).
   355→  if (!isUserSession) {
   356→    const trips = localTrips ?? [];
   357→    return (
   358→      <main className="flex-1 flex flex-col bg-muted/30">
   359→        <section className="flex-1 mx-auto w-full max-w-5xl px-4 py-8 sm:py-10">
   360→          <h1 className="text-2xl font-bold sm:text-3xl">{L.title[lang]}</h1>
   361→          <p className="mt-1 text-sm text-muted-foreground">
   362→            {L.guest.subtitleA[lang]}{" "}
   363→            <strong>{L.guest.subtitleStrong[lang]}</strong>{" "}
   364→            {L.guest.subtitleB[lang]}
   365→          </p>
   366→
   367→          {/* TASK 8 / D8-E: account vrstica — gumb "Prijavi se" preložen iz
   368→              odstranjenega lastnega headerja (logotip → Navigation). */}
   369→          <div className="mt-3">
   370→            <Button asChild size="sm" className="gap-1.5 h-10">
   371→              <Link href="/prijava">{L.guest.signIn[lang]}</Link>
   372→            </Button>
   373→          </div>
   374→
   375→          {/* Iskrena razlaga lokalnega shranjevanja + ponudba sinhronizacije */}
   376→          <Alert className="mt-6 border-primary/30 bg-primary/5">
   377→            <CloudUpload className="size-4 text-primary" aria-hidden="true" />
   378→            <AlertTitle>{L.guest.alertTitle[lang]}</AlertTitle>
   379→            <AlertDescription className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   380→              <span className="text-sm">
   381→                {L.guest.alertText[lang]}
   382→              </span>
   383→              <Button asChild size="sm" className="gap-1.5 shrink-0">
   384→                <Link href="/prijava">
   385→                  {L.guest.createAccount[lang]}
   386→                  <ArrowRight className="size-3.5" aria-hidden="true" />
   387→                </Link>
   388→              </Button>
   389→            </AlertDescription>
   390→          </Alert>
   391→
   392→          {/* TASK 8 / D8-B: zbirka "Moja pot" — odkrivanje → dodaj → nadaljuj */}
   393→          <MyTripView className="mt-8" />
   394→
   395→          {/* Lokalna potovanja (iste kartice kot prijavljeni) */}
   396→          <section aria-labelledby="lokalna-potovanja" className="mt-8">
   397→            <div className="flex items-center justify-between gap-3">
   398→              <h2
   399→                id="lokalna-potovanja"
   400→                className="flex items-center gap-2 text-lg font-semibold"
   401→              >
   402→                <HardDrive className="size-5 text-primary" aria-hidden="true" />
   403→                {L.trips.heading[lang]}
   404→              </h2>
   405→              <Badge variant="secondary">
   406→                {L.trips.count[lang](trips.length)}
   407→              </Badge>
   408→            </div>
   409→
   410→            {localTrips === null ? (
   411→              /* TASK 8 / F3-B: skeleton vrstice družine (status nosi vrstica,
   412→                  skeleti so dekorativni) — prej pulse bloki s hardcoded
   413→                  slovensko aria oznako. */
   414→              <LoadingState variant="block" rows={2} className="mt-4 items-stretch" />
   415→            ) : trips.length === 0 ? (
   416→              <EmptyState
   417→                icon={Map}
   418→                title={L.trips.emptyGuestTitle[lang]}
   419→                description={L.trips.emptyGuestDesc[lang]}
   420→                action={{ label: L.trips.emptyAction[lang], href: "/nacrtuj" }}
   421→                className="mt-4"
   422→              />
   423→            ) : (
   424→              <ul className="mt-4 grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   425→                {trips.map((trip) => (
   426→                  <li key={trip.shareId}>
   427→                    <Card className="h-full transition-shadow hover:shadow-md">
   428→                      <CardHeader className="pb-2">
   429→                        <CardTitle className="text-base line-clamp-1">
   430→                          {trip.name ??
   431→                            L.trips.fallbackName[lang](
   432→                              formatDate(trip.savedAt, lang)
   433→                            )}
   434→                        </CardTitle>
   435→                        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   436→                          <span className="inline-flex items-center gap-1">
   437→                            <HardDrive className="size-3.5" aria-hidden="true" />
   438→                            {L.guest.onThisDevice[lang]}
   439→                          </span>
   440→                        </div>
   441→                      </CardHeader>
   442→                      <CardContent className="flex items-center justify-between gap-3 pt-0">
   443→                        <span className="text-xs text-muted-foreground">
   444→                          {L.trips.saved[lang](formatDate(trip.savedAt, lang))}
   445→                        </span>
   446→                        <Button asChild size="sm" className="gap-1.5 shrink-0">
   447→                          <Link href={`/pot/${trip.shareId}`}>
   448→                            {L.open[lang]}
   449→                            <ArrowRight className="size-3.5" aria-hidden="true" />
   450→                          </Link>
   451→                        </Button>
   452→                      </CardContent>
   453→                    </Card>
   454→                  </li>
   455→                ))}
   456→              </ul>
   457→            )}
