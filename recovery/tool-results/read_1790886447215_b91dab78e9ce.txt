   102→  try {
   103→    return new Intl.DateTimeFormat(lang === "en" ? "en-GB" : "sl-SI", {
   104→      day: "numeric",
   105→      month: "long",
   106→      year: "numeric",
   107→    }).format(new Date(iso));
   108→  } catch {
   109→    return iso.slice(0, 10);
   110→  }

   245→export function MojaPotovanjaView() {
   246→  const router = useRouter();
   247→  const { toast } = useToast();
   248→  const { data: session, status } = useSession();
   249→  // F4-C: jezik chroma (SL privzet — EN le, kadar locale zahteva; P4-8).
   250→  const locale = useLocale();
   251→  const lang: "sl" | "en" = locale === "en" ? "en" : "sl";
   252→
   253→  const [data, setData] = useState<TripsResponse | null>(null);
   254→  const [loadError, setLoadError] = useState<string | null>(null);
   255→  // ISSUE #22 §30.C: Nadaljuj na poti — nalagalnik (per shareId) za
   256→  // prenos shranjene poti v Go Mode (naprava) + preusmeritev na /na-poti.
   257→  const [resumingTripId, setResumingTripId] = useState<string | null>(null);
   258→  const [resending, setResending] = useState(false);
   259→  // TASK 4 / K-6: gostova LOKALNA potovanja (hidrirajo se TEKOM mounta —
   260→  // localStorage je klient-only, da ni hydration mismatcha).
   261→  const [localTrips, setLocalTrips] = useState<TrackedTrip[] | null>(null);
   262→
   263→  const isUserSession =
   264→    status === "authenticated" && session?.user?.accountType === "user";
   265→
   266→  // TASK 4 / K-6 (UX FIX PASS): NEPREUSMERJENI gost — prej je klik
   267→  // "Moja potovanja" gosta vrgel na /prijava BREZ sporočila, da je pravkar
   268→  // shranjen načrt DEJANSKO še dostopen (deljiva povezava + localStorage).
   269→  // Prelom zlate poti točno pri MY TRIP (živi dokaz revizije, korak 10).
   270→  // Zdaj: gost vidi SVOJA lokalna potovanja + iskreno ponudbo računa.
   271→  // B2B seja še vedno nima kaj iskati tukaj — nazaj na domačo stran.
   272→  useEffect(() => {
   273→    if (
   274→      status === "authenticated" &&
   275→      session?.user?.accountType !== "user"
   276→    ) {
   277→      router.replace("/");
   278→    }
   279→  }, [status, session, router]);
   280→
   281→  // K-6: hidratacija lokalnih potovanj (samo gost — prijavljeni uporablja
   282→  // strežniški seznam iz /api/user/trips + claim plast pri prijavi).
   283→  useEffect(() => {
   284→    if (isUserSession) return;
   285→    const hydrate = setTimeout(() => setLocalTrips(getSavedTrips()), 0);
   286→    return () => clearTimeout(hydrate);
   287→  }, [isUserSession]);
   288→
   289→  // Nalaganje podatkov (samo za B2C sejo)
   290→  useEffect(() => {
   291→    if (!isUserSession) return;
   292→    let cancelled = false;
   293→
   294→    // TASK 8 / F2-A: pull sinhronizacija zbirke "Moja pot" — prinese
   295→    // predmete z drugih naprav (union-merge v localStorage; MyTripView se
   296→    // osveži prek dogodka dai:my-trip-changed). Neblokirajoče + fail-open.
   297→    void syncMyTripToServer();
   298→
   299→    fetch("/api/user/trips", { cache: "no-store" })
   300→      .then(async (r) => {
   301→        if (!r.ok) {
   302→          const d = await r.json().catch(() => ({}));
   303→          throw new Error(d?.error ?? L.errors.loadFailed[lang]);
   304→        }
   305→        return r.json() as Promise<TripsResponse>;
   306→      })
   307→      .then((d) => {
   308→        if (!cancelled) setData(d);
   309→      })
   310→      .catch((e) => {
   311→        if (!cancelled) {
   312→          setLoadError(
   313→            e instanceof Error ? e.message : L.errors.loadFailed[lang]
   314→          );
   315→        }
   316→      });
   317→
   318→    return () => {
   319→      cancelled = true;
   320→    };
   321→  }, [isUserSession, lang]);
   322→
   323→  // Ponovno pošiljanje verifikacijske povezave
   324→  const resendVerification = async () => {
   325→    setResending(true);
   326→    try {
   327→      const res = await fetch("/api/user/verify-email", {
   328→        method: "POST",
   329→        headers: { "Content-Type": "application/json" },
   330→        body: JSON.stringify({ action: "request" }),
   331→      });
   332→      const d = await res.json().catch(() => ({}));
   333→      if (!res.ok) throw new Error(d?.error ?? L.errors.sendFailed[lang]);
   334→      toast({
   335→        title: L.toast.linkSent[lang],
   336→        description:
   337→          d?.message ??
   338→          L.toast.checkInbox[lang],
   339→      });
   340→    } catch (err) {
   341→      toast({
   342→        variant: "destructive",
   343→        title: L.errorTitle[lang],
   344→        description: err instanceof Error ? err.message : L.toast.tryAgain[lang],
   345→      });
   346→    } finally {
   347→      setResending(false);
   348→    }
   349→  };
   350→
   351→  /** ISSUE #22 §30.C: Nadaljuj na poti — pridobi shranjeno pot, zgradi Go
   352→   *  Mode zapis (MyTripView na napravi) in preusmeri na /na-poti. Napaka
   353→   *  (zasebna pot/dostop) → iskren odprt potn strani (tam je poln tok). */
   354→  const resumeTrip = async (shareId: string, fallbackName: string) => {
   355→    if (resumingTripId) return;
   356→    setResumingTripId(shareId);
   357→    try {
   358→      const res = await fetch(
   359→        `/api/itinerary/shared/${encodeURIComponent(shareId)}`
   360→      );
   361→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   362→      const data = (await res.json()) as {
   363→        name?: string | null;
   364→        itinerary?: unknown;
   365→      };
   366→      if (data.itinerary == null) throw new Error("no itinerary");
   367→      const view = buildItineraryGoView(data.itinerary as never, {
   368→        lang: "sl",
   369→        name: data.name ?? fallbackName,
   370→      });
   371→      if (saveItineraryGoTrip(view, { shareId })) {
   372→        router.push(lang === "sl" ? "/na-poti" : "/en/na-poti");
   373→        return;
   374→      }
   375→      throw new Error("persist failed");
   376→    } catch {
   377→      toast({ description: L.resumeError[lang] });
   378→      router.push(`/pot/${shareId}`);
   379→    } finally {
   380→      setResumingTripId(null);
   381→    }
   382→  };
   383→
   384→  // Spinner SAMO med nalaganjem seje (gost ne čaka več na preusmeritev —
   385→  // TASK 4 / K-6: takoj dobi svoj pogled). TASK 8 / F3-B: LoadingState
   386→  // (block) — oznaka je L-pattern SL/EN (družinski privzetek).
   387→  if (status === "loading" || (status === "authenticated" && !isUserSession)) {
   388→    return (
   389→      <main
   390→        className="flex-1 flex items-center justify-center py-24"
   391→        aria-busy="true"
   392→      >
   393→        <LoadingState variant="block" />
   394→      </main>
   395→    );
   396→
   397→  }
   398→
   399→  // TASK 4 / K-6: GOSTOV POGLED — lokalna potovanja te naprave + iskrena
   400→  // ponudba računa (sinhronizacija). Ni login zida: načrti so DEJANSKO
   401→  // dostopni (deljive povezave so javne; localStorage nosi seznam).
   402→  if (!isUserSession) {
   403→    const trips = localTrips ?? [];
   404→    return (
   405→      <main className="flex-1 flex flex-col bg-muted/30">
   406→        <section className="flex-1 mx-auto w-full max-w-5xl px-4 py-8 sm:py-10">
   407→          <h1 className="text-2xl font-bold sm:text-3xl">{L.title[lang]}</h1>
   408→          <p className="mt-1 text-sm text-muted-foreground">
   409→            {L.guest.subtitleA[lang]}{" "}
   410→            <strong>{L.guest.subtitleStrong[lang]}</strong>{" "}
   411→            {L.guest.subtitleB[lang]}
   412→          </p>
   413→
   414→          {/* TASK 8 / D8-E: account vrstica — gumb "Prijavi se" preložen iz
   415→              odstranjenega lastnega headerja (logotip → Navigation). */}
   416→          <div className="mt-3">
   417→            <Button asChild size="sm" className="gap-1.5 h-10">
   418→              <Link href="/prijava">{L.guest.signIn[lang]}</Link>
   419→            </Button>
   420→          </div>
   421→
   422→          {/* Iskrena razlaga lokalnega shranjevanja + ponudba sinhronizacije */}
   423→          <Alert className="mt-6 border-primary/30 bg-primary/5">
   424→            <CloudUpload className="size-4 text-primary" aria-hidden="true" />
   425→            <AlertTitle>{L.guest.alertTitle[lang]}</AlertTitle>
   426→            <AlertDescription className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   427→              <span className="text-sm">
   428→                {L.guest.alertText[lang]}
   429→              </span>
   430→              <Button asChild size="sm" className="gap-1.5 shrink-0">
   431→                <Link href="/prijava">
   432→                  {L.guest.createAccount[lang]}
   433→                  <ArrowRight className="size-3.5" aria-hidden="true" />
   434→                </Link>
   435→              </Button>
   436→            </AlertDescription>
   437→          </Alert>
   438→
   439→          {/* TASK 8 / D8-B: zbirka "Moja pot" — odkrivanje → dodaj → nadaljuj */}
   440→          <MyTripView className="mt-8" />
   441→
   442→          {/* Lokalna potovanja (iste kartice kot prijavljeni) */}
   443→          <section aria-labelledby="lokalna-potovanja" className="mt-8">
   444→            <div className="flex items-center justify-between gap-3">
   445→              <h2
   446→                id="lokalna-potovanja"
