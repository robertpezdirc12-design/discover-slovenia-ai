    84→function slUnit(
    85→  n: number,
    86→  one: string,
    87→  two: string,
    88→  few: string,
    89→  many: string
    90→): string {
    91→  if (n === 1) return one;
    92→  const r = n % 100;
    93→  if (r === 2) return two;
    94→  if (r === 3 || r === 4) return few;
    95→  return many;
    96→}

    98→/** "pred 2 min", "pred 3 h" — relativni čas z varnostnim stropom. */
    99→function slTimeAgo(iso: string, floorIso?: string): string {
   100→  const ms = Date.parse(iso);
   101→  if (Number.isNaN(ms)) return "neznano";
   102→
   103→  // Sanity cap: anketa ne more biti starejša od potovanja (pokvarjeni podatki)
   104→  if (floorIso) {
   105→    const floor = Date.parse(floorIso);
   106→    if (!Number.isNaN(floor) && ms < floor) {
   107→      return new Date(ms).toLocaleDateString("sl-SI", {
   108→        day: "numeric",
   109→        month: "short",
   110→        year: "numeric",
   111→      });
   112→    }
   113→  }
   114→
   115→  const seconds = Math.max(0, Math.floor((Date.now() - ms) / 1000));
   116→  if (seconds < 45) return "pravkar";
   117→
   118→  const minutes = Math.floor(seconds / 60);
   119→  if (minutes < 60) {
   120→    return `pred ${minutes} ${slUnit(minutes, "minuto", "minutama", "minutami", "minutami")}`;
   121→  }
   122→  const hours = Math.floor(minutes / 60);
   123→  if (hours < 24) {
   124→    return `pred ${hours} ${slUnit(hours, "uro", "urama", "urami", "urami")}`;
   125→  }
   126→  const days = Math.floor(hours / 24);
   127→  if (days < 30) {
   128→    return `pred ${days} ${slUnit(days, "dnem", "dnevoma", "dnevi", "dnevi")}`;
   129→  }
   130→  return new Date(ms).toLocaleDateString("sl-SI", {
   131→    day: "numeric",
   132→    month: "short",
   133→    year: "numeric",
   134→  });
   135→}

   137→/** "3 glasovi" — pravilne slovenske oblike. */
   138→function votesLabel(n: number): string {
   139→  if (n === 0) return "Ni še glasov";
   140→  return `${n} ${slUnit(n, "glas", "glasa", "glasi", "glasov")}`;
   141→}

   146→export function TripPolls({ shareId, initialPolls, createdAt }: TripPollsProps) {
   147→  const { toast } = useToast();
   148→  // TASK 8 / F3-B: jezik za L-pattern placeholder nalaganja (SL privzeto).
   149→  const locale = useLocale();
   150→  const lang: "sl" | "en" = locale === "en" ? "en" : "sl";
   151→
   152→  const [clientId, setClientId] = useState<string>("");
   153→  const [mounted, setMounted] = useState<boolean>(false);
   154→  const [polls, setPolls] = useState<TripPollItem[]>(() =>
   155→    (initialPolls ?? []).map((p) => ({ ...p, myVote: null, isAuthor: false }))
   156→  );
   157→
   158→  // Obrazec za novo anketo
   159→  const [formOpen, setFormOpen] = useState<boolean>(false);
   160→  const [question, setQuestion] = useState<string>("");
   161→  const [options, setOptions] = useState<string[]>(["", ""]);
   162→  const [authorName, setAuthorName] = useState<string>("");
   163→  const [creating, setCreating] = useState<boolean>(false);
   164→
   165→  // Za katero anketo trenutno poteka oddaja glasu (pollId → optionIdx)
   166→  const [votePending, setVotePending] = useState<Record<string, number>>({});
   167→  const [actionPending, setActionPending] = useState<string | null>(null);
   168→
   169→  /** Znova pridobi ankete z mojim glasom (GET /api/poll?voterId=). */
   170→  const refreshPolls = useCallback(
   171→    async (cid: string) => {
   172→      try {
   173→        const res = await fetch(
   174→          `/api/poll?shareId=${encodeURIComponent(shareId)}&voterId=${encodeURIComponent(cid)}`,
   175→          { cache: "no-store" }
   176→        );
   177→        if (!res.ok) return;
   178→        const data = (await res.json().catch(() => null)) as {
   179→          polls?: TripPollItem[];
   180→        } | null;
   181→        if (Array.isArray(data?.polls)) {
   182→          setPolls(data.polls);
   183→        }
   184→      } catch {
   185→        // Off-line / napaka — ohranimo trenutno stanje
   186→      }
   187→    },
   188→    [shareId]
   189→  );
   190→
   191→  // Mount: clientId (isti ID kot ostale socialne funkcije) + prihranjeno ime
   192→  useEffect(() => {
   193→    setMounted(true);
   194→    if (!shareId) return;
   195→    try {
   196→      // 99-b: read-or-create anonimni ID (private mode → null → stanje
   197→      // ostane nedotaknjeno, ankete ostanejo berljive)
   198→      const cid = getVoterId();
   199→      if (cid) {
   200→        setClientId(cid);
   201→
   202→        // Prihranjeno ime (skupno s komentarji)
   203→        const savedName = getAuthorName() ?? "";
   204→        if (savedName) setAuthorName(savedName);
   205→
   206→        // Dopolni myVote/isAuthor (server-side podatki jih ne poznajo)
   207→        void refreshPolls(cid);
   208→      }
   209→    } catch {
   210→      // private mode — ankete ostanejo berljive, brez mojega glasu
   211→    }
   212→  }, [shareId, refreshPolls]);
   213→
   214→  // ========================================================================
   215→  // Ustvari anketo
   216→  // ========================================================================
   217→  const submitPoll = useCallback(
   218→    async (e: React.FormEvent<HTMLFormElement>) => {
   219→      e.preventDefault();
   220→      if (creating || !shareId || !clientId) return;
   221→
   222→      const q = question.trim();
   223→      const opts = options.map((o) => o.trim()).filter((o) => o.length > 0);
   224→
   225→      if (q.length < 2 || q.length > QUESTION_MAX) {
   226→        toast({
   227→          title: "Vprašanje ni ustrezno",
   228→          description: `Vprašanje mora imeti med 2 in ${QUESTION_MAX} znakov.`,
   229→          variant: "destructive",
   230→        });
   231→        return;
   232→      }
   233→      if (opts.length < OPTIONS_MIN || opts.length > OPTIONS_MAX) {
   234→        toast({
   235→          title: "Anketa potrebuje možnosti",
   236→          description: `Vpiši vsaj ${OPTIONS_MIN} (največ ${OPTIONS_MAX}) neprazni možnosti.`,
   237→          variant: "destructive",
   238→        });
   239→        return;
   240→      }
   241→      if (opts.some((o) => o.length > OPTION_MAX)) {
   242→        toast({
   243→          title: "Možnost je predolga",
   244→          description: `Posamezna možnost je lahko dolga največ ${OPTION_MAX} znakov.`,
   245→          variant: "destructive",
   246→        });
   247→        return;
   248→      }
   249→
   250→      setCreating(true);
   251→      try {
   252→        const res = await fetch("/api/poll", {
   253→          method: "POST",
   254→          headers: { "Content-Type": "application/json" },
   255→          body: JSON.stringify({
   256→            shareId,
   257→            question: q,
   258→            options: opts,
   259→            authorName: authorName.trim(),
   260→            clientId,
   261→          }),
   262→        });
   263→        const data: unknown = await res.json().catch(() => null);
   264→        const d = data as { success?: unknown; poll?: unknown } | null;
   265→
   266→        if (!res.ok || d?.success !== true || !d?.poll) {
   267→          const error = (data as { error?: unknown } | null)?.error;
   268→          throw new Error(
   269→            typeof error === "string" ? error : `HTTP ${res.status}`
   270→          );
   271→        }
   272→
   273→        const poll = d.poll as TripPollItem;
   274→        setPolls((prev) => [poll, ...prev]);
   275→        setQuestion("");
   276→        setOptions(["", ""]);
   277→        setFormOpen(false);
   278→
   279→        // Prihrani ime (skupno s komentarji; defenzivno — private mode
   280→        // se mirno preskoči znotraj lib-a)
   281→        saveAuthorName(authorName.trim());
   282→
   283→        toast({
   284→          title: "Anketa je ustvarjena",
   285→          description: "Povezavo deli s skupino in glasujte skupaj!",
   286→        });
   287→      } catch (err) {
   288→        const message = err instanceof Error ? err.message : "Poskusi znova.";
   289→        toast({
   290→          title: "Ankete ni bilo mogoče ustvariti",
   291→          description: message,
   292→          variant: "destructive",
   293→        });
   294→      } finally {
   295→        setCreating(false);
   296→      }
   297→    },
   298→    [creating, shareId, clientId, question, options, authorName, toast]
   299→  );
   300→
   301→  // ========================================================================
   302→  // Glasuj / prestavi glas (optimistično z revertom)
   303→  // ========================================================================
   304→  const castVote = useCallback(
   305→    async (pollId: string, optionIdx: number) => {
   306→      if (!clientId || !shareId) return;
   307→      const poll = polls.find((p) => p.id === pollId);
   308→      if (!poll || poll.closed) return;
   309→      if (votePending[pollId] !== undefined) return; // že v teku
   310→      if (poll.myVote === optionIdx) return; // klik na svoj glas = no-op
   311→
   312→      // Optimistični update: prestavi moj glas med opcijami
   313→      const prev = poll;
   314→      const next: TripPollItem = {
   315→        ...poll,
   316→        counts: poll.counts.map((c, i) => {
   317→          if (i === optionIdx) return c + 1;
   318→          if (poll.myVote === i) return Math.max(0, c - 1);
   319→          return c;
   320→        }),
   321→        total: poll.myVote === null ? poll.total + 1 : poll.total,
   322→        myVote: optionIdx,
   323→      };
   324→      setPolls((all) => all.map((p) => (p.id === pollId ? next : p)));
   325→      setVotePending((v) => ({ ...v, [pollId]: optionIdx }));
   326→
   327→      try {
   328→        const res = await fetch("/api/poll/vote", {
   329→          method: "POST",
   330→          headers: { "Content-Type": "application/json" },
   331→          body: JSON.stringify({ pollId, voterId: clientId, optionIdx }),
   332→        });
   333→        const data: unknown = await res.json().catch(() => null);
   334→        const d = data as
   335→          | { success?: unknown; counts?: unknown; total?: unknown; myVote?: unknown }
   336→          | null;
   337→
   338→        if (
   339→          !res.ok ||
   340→          d?.success !== true ||
   341→          !Array.isArray(d?.counts) ||
   342→          typeof d?.total !== "number"
   343→        ) {
   344→          throw new Error(
   345→            (data as { error?: unknown } | null)?.error instanceof Error
   346→              ? String((data as { error?: unknown }).error)
   347→              : `HTTP ${res.status}`
   348→          );
   349→        }
   350→
   351→        // Strežniško stanje je resnica (ali izgladi lokalne razlike)
   352→        setPolls((all) =>
   353→          all.map((p) =>
   354→            p.id === pollId
   355→              ? {
   356→                  ...p,
   357→                  counts: d.counts as number[],
   358→                  total: d.total as number,
   359→                  myVote:
   360→                    typeof d.myVote === "number" ? (d.myVote as number) : null,
   361→                }
   362→              : p
   363→          )
   364→        );
   365→      } catch (err) {
   366→        // Revert na prejšnje stanje
   367→        setPolls((all) => all.map((p) => (p.id === pollId ? prev : p)));
   368→        const message = err instanceof Error ? err.message : "Poskusi znova.";
   369→        toast({
   370→          title: "Glasa ni bilo mogoče oddati",
   371→          description: message,
   372→          variant: "destructive",
   373→        });
   374→      } finally {
   375→        setVotePending((v) => {
   376→          const copy = { ...v };
   377→          delete copy[pollId];
   378→          return copy;
   379→        });
   380→      }
   381→    },
   382→    [clientId, shareId, polls, votePending, toast]
   383→  );
   384→
   385→  // ========================================================================
   386→  // Zaključi / ponovno odpri (samo avtor)
   387→  // ========================================================================
   388→  const toggleClose = useCallback(
   389→    async (pollId: string, closed: boolean) => {
   390→      if (!clientId || actionPending) return;
   391→      setActionPending(pollId);
   392→      try {
   393→        const res = await fetch("/api/poll", {
   394→          method: "PATCH",
   395→          headers: { "Content-Type": "application/json" },
   396→          body: JSON.stringify({ pollId, clientId, closed }),
   397→        });
   398→        const data: unknown = await res.json().catch(() => null);
   399→        const d = data as { success?: unknown; poll?: unknown } | null;
   400→        if (!res.ok || d?.success !== true || !d?.poll) {
   401→          throw new Error(
   402→            (data as { error?: unknown } | null)?.error instanceof Error
   403→              ? String((data as { error?: unknown }).error)
   404→              : `HTTP ${res.status}`
   405→          );
   406→        }
   407→        const poll = d.poll as TripPollItem;
   408→        setPolls((all) => all.map((p) => (p.id === pollId ? poll : p)));
   409→        toast({
   410→          title: closed ? "Anketa je zaključena" : "Anketa je znova odprta",
   411→          description: closed
   412→            ? "Rezultati so vidni, glasovanje pa zaprto."
   413→            : "Skupina lahko spet glasuje.",
   414→        });
   415→      } catch (err) {
   416→        const message = err instanceof Error ? err.message : "Poskusi znova.";
   417→        toast({
   418→          title: "Napaka pri posodabljanju ankete",
   419→          description: message,
   420→          variant: "destructive",
   421→        });
   422→      } finally {
   423→        setActionPending(null);
   424→      }
   425→    },
   426→    [clientId, actionPending, toast]
   427→  );
   428→
   429→  // ========================================================================
   430→  // Izbriši (samo avtor)
   431→  // ========================================================================
   432→  const deletePoll = useCallback(
   433→    async (pollId: string) => {
   434→      if (!clientId || actionPending) return;
   435→      setActionPending(pollId);
   436→      try {
   437→        const res = await fetch("/api/poll", {
   438→          method: "DELETE",
   439→          headers: { "Content-Type": "application/json" },
   440→          body: JSON.stringify({ pollId, clientId }),
   441→        });
   442→        const data: unknown = await res.json().catch(() => null);
   443→        const d = data as { success?: unknown } | null;
   444→        if (!res.ok || d?.success !== true) {
   445→          throw new Error(
   446→            (data as { error?: unknown } | null)?.error instanceof Error
   447→              ? String((data as { error?: unknown }).error)
   448→              : `HTTP ${res.status}`
   449→          );
   450→        }
   451→        setPolls((all) => all.filter((p) => p.id !== pollId));
   452→        toast({ title: "Anketa je izbrisana" });
   453→      } catch (err) {
   454→        const message = err instanceof Error ? err.message : "Poskusi znova.";
   455→        toast({
   456→          title: "Ankete ni bilo mogoče izbrisati",
   457→          description: message,
   458→          variant: "destructive",
   459→        });
   460→      } finally {
   461→        setActionPending(null);
   462→      }
   463→    },
   464→    [clientId, actionPending, toast]
   465→  );
   466→
   467→  const openCount = useMemo(
   468→    () => polls.filter((p) => !p.closed).length,
   469→    [polls]
   470→  );
   471→
   472→  // ========================================================================
   473→  // RENDER
   474→  // ========================================================================
   475→  return (
   476→    <Card className="print-hide print:hidden overflow-hidden">
   477→      <CardContent className="p-4 sm:p-6">
   478→        <div className="mb-4 flex flex-wrap items-center justify-between gap-3">
   479→          <div className="flex items-center gap-2">
   480→            <Vote className="size-5 text-primary" aria-hidden="true" />
   481→            <h2 className="text-lg font-semibold">Skupinske ankete</h2>
   482→            <span className="text-sm text-muted-foreground">
   483→              {openCount > 0
   484→                ? `${openCount} ${slUnit(openCount, "odprta", "odprti", "odprte", "odprtih")}`
   485→                : polls.length > 0
   486→                  ? "vse zaključene"
   487→                  : "še ni anket"}
   488→            </span>
   489→          </div>
   490→          <Button
   491→            type="button"
   492→            variant={formOpen ? "outline" : "default"}
   493→            size="sm"
   494→            onClick={() => setFormOpen((o) => !o)}
   495→            aria-expanded={formOpen}
   496→            aria-controls="poll-form"
   497→            className="gap-1.5"
   498→          >
   499→            {formOpen ? (
   500→              <>
   501→                <X className="size-4" aria-hidden="true" /> Zapri
   502→              </>
   503→            ) : (
   504→              <>
   505→                <Plus className="size-4" aria-hidden="true" /> Nova anketa
   506→              </>
   507→            )}
   508→          </Button>
   509→        </div>
   510→
   511→        <p className="mb-4 text-sm text-muted-foreground">
   512→          Odločite se skupaj: postavite vprašanje (npr. &bdquo;Kateri dan
   513→          odpotujemo?&ldquo;) in vsak član skupine odda en glas. Glas lahko
   514→          prestavite, dokler je anketa odprta.
   515→        </p>
   516→
   517→        {/* === Obrazec za novo anketo === */}
   518→        {formOpen && (
   519→          <form
   520→            id="poll-form"
   521→            onSubmit={(e) => void submitPoll(e)}
   522→            className="mb-6 space-y-3 rounded-lg border border-border bg-muted/30 p-4"
   523→          >
   524→            <Input
   525→              value={question}
   526→              onChange={(e) => setQuestion(e.target.value)}
   527→              placeholder="Vprašanje za skupino (npr. Kateri dan odpotujemo?)"
   528→              required
   529→              maxLength={QUESTION_MAX}
   530→              minLength={2}
   531→              aria-label="Vprašanje ankete"
   532→              className="h-11"
   533→              disabled={creating}
   534→            />
   535→
   536→            <div className="space-y-2">
   537→              {options.map((opt, i) => (
   538→                <div key={`opt-${i}`} className="flex items-center gap-2">
   539→                  <span
   540→                    className="flex size-6 shrink-0 items-center justify-center rounded-full bg-primary/10 text-xs font-bold text-primary"
   541→                    aria-hidden="true"
   542→                  >
   543→                    {String.fromCharCode(65 + i)}
   544→                  </span>
   545→                  <Input
   546→                    value={opt}
   547→                    onChange={(e) =>
   548→                      setOptions((prev) =>
   549→                        prev.map((o, j) => (j === i ? e.target.value : o))
   550→                      )
   551→                    }
   552→                    placeholder={`Možnost ${String.fromCharCode(65 + i)}`}
   553→                    maxLength={OPTION_MAX}
   554→                    aria-label={`Možnost ${String.fromCharCode(65 + i)}`}
   555→                    className="h-11"
   556→                    disabled={creating}
   557→                  />
   558→                  {options.length > OPTIONS_MIN && (
   559→                    <Button
   560→                      type="button"
   561→                      variant="ghost"
   562→                      size="icon"
   563→                      onClick={() =>
   564→                        setOptions((prev) => prev.filter((_, j) => j !== i))
   565→                      }
   566→                      aria-label={`Odstrani možnost ${String.fromCharCode(65 + i)}`}
   567→                      disabled={creating}
   568→                      className="size-9 shrink-0"
   569→                    >
   570→                      <X className="size-4" aria-hidden="true" />
   571→                    </Button>
   572→                  )}
   573→                </div>
   574→              ))}
   575→              {options.length < OPTIONS_MAX && (
   576→                <Button
   577→                  type="button"
   578→                  variant="outline"
   579→                  size="sm"
   580→                  onClick={() => setOptions((prev) => [...prev, ""])}
   581→                  disabled={creating}
   582→                  className="gap-1.5"
   583→                >
   584→                  <Plus className="size-4" aria-hidden="true" /> Dodaj možnost
   585→                </Button>
   586→              )}
   587→            </div>
   588→
   589→            <Input
   590→              value={authorName}
   591→              onChange={(e) => setAuthorName(e.target.value)}
   592→              placeholder="Tvoje ime (opcijsko — vidno ob anketi)"
   593→              maxLength={AUTHOR_NAME_MAX}
   594→              aria-label="Tvoje ime"
   595→              className="h-11"
   596→              disabled={creating}
   597→            />
   598→
   599→            <div className="flex items-center justify-between gap-3">
   600→              <span className="text-xs text-muted-foreground">
   601→                {question.trim().length}/{QUESTION_MAX}
   602→              </span>
   603→              <Button type="submit" disabled={creating} className="gap-1.5">
   604→                {creating ? (
   605→                  <Loader2
   606→                    className="size-4 animate-spin"
   607→                    aria-hidden="true"
   608→                  />
   609→                ) : (
   610→                  <BarChart3 className="size-4" aria-hidden="true" />
   611→                )}
   612→                Ustvari anketo
   613→              </Button>
   614→            </div>
   615→          </form>
   616→        )}
   617→
   618→        {/* === Seznam anket === */}
   619→        {polls.length === 0 ? (
   620→          <p className="rounded-lg border border-dashed border-border p-6 text-center text-sm text-muted-foreground">
   621→            {mounted
   622→              ? "Še ni nobene ankete — postavi prvo vprašanje skupini!"
   623→              : L.loadingPolls[lang]}
   624→          </p>
   625→        ) : (
   626→          <ul className="space-y-4" aria-live="polite">
   627→            {polls.map((poll) => (
   628→              <li
   629→                key={poll.id}
   630→                className="rounded-lg border border-border p-4"
   631→              >
   632→                <PollCard
   633→                  poll={poll}
   634→                  mounted={mounted}
   635→                  tripCreatedAt={createdAt}
   636→                  votePendingIdx={
   637→                    votePending[poll.id] !== undefined
   638→                      ? votePending[poll.id]
   639→                      : null
   640→                  }
   641→                  actionPending={actionPending === poll.id}
   642→                  onVote={(idx) => void castVote(poll.id, idx)}
   643→                  onToggleClose={() =>
   644→                    void toggleClose(poll.id, !poll.closed)
   645→                  }
   646→                  onDelete={() => void deletePoll(poll.id)}
   647→                />
   648→              </li>
   649→            ))}
   650→          </ul>
   651→        )}
   652→      </CardContent>
   653→    </Card>
   654→  );
   655→}

   660→interface PollCardProps {
   661→  poll: TripPollItem;
   662→  mounted: boolean;
   663→  tripCreatedAt: string;
   664→  /** Index opcije, katere glas je v teku (null = nič). */
   665→  votePendingIdx: number | null;
   666→  actionPending: boolean;
   667→  onVote: (optionIdx: number) => void;
   668→  onToggleClose: () => void;
   669→  onDelete: () => void;
   670→}
   671→
   672→function PollCard({
   673→  poll,
   674→  mounted,
   675→  tripCreatedAt,
   676→  votePendingIdx,
   677→  actionPending,
   678→  onVote,
   679→  onToggleClose,
   680→  onDelete,
   681→}: PollCardProps) {
   682→  const maxCount = Math.max(1, ...poll.counts);
   683→  const total = poll.total;
   684→
   685→  return (
   686→    <div>
   687→      {/* Vprašanje + meta */}
   688→      <div className="mb-1 flex items-start justify-between gap-2">
   689→        <h3 className="font-semibold leading-snug">{poll.question}</h3>
   690→        {poll.closed && (
   691→          <span className="inline-flex shrink-0 items-center gap-1 rounded-full bg-muted px-2.5 py-1 text-xs font-medium text-muted-foreground">
   692→            <Lock className="size-3" aria-hidden="true" />
   693→            Zaključena
   694→          </span>
   695→        )}
   696→      </div>
   697→      <p className="mb-3 text-xs text-muted-foreground">
   698→        {poll.authorName ? `${poll.authorName} · ` : ""}
   699→        {mounted ? slTimeAgo(poll.createdAt, tripCreatedAt) : ""}
   700→      </p>
   701→
   702→      {/* Možnosti — klikljive vrstice z rezultati */}
   703→      <div className="space-y-2">
   704→        {poll.options.map((opt, i) => {
   705→          const count = poll.counts[i] ?? 0;
   706→          const pct = total > 0 ? Math.round((count / total) * 100) : 0;
   707→          const isMyVote = poll.myVote === i;
   708→          const isPending = votePendingIdx === i;
   709→          const widthPct = total > 0 ? Math.round((count / maxCount) * 100) : 0;
   710→
   711→          return (
   712→            <button
   713→              key={`${poll.id}-opt-${i}`}
   714→              type="button"
   715→              onClick={() => onVote(i)}
   716→              disabled={poll.closed || isPending}
   717→              aria-pressed={isMyVote}
   718→              aria-label={`Glasuj za: ${opt} (${count} ${slUnit(count, "glas", "glasa", "glasi", "glasov")})`}
   719→              className={cn(
   720→                "relative flex w-full items-center gap-2.5 overflow-hidden rounded-lg border px-3 py-2.5 text-left text-sm transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 disabled:cursor-default",
   721→                !poll.closed &&
   722→                  "hover:border-primary/50 focus-visible:border-primary/50",
   723→                isMyVote
   724→                  ? "border-primary bg-primary/5 font-medium"
   725→                  : "border-border bg-background",
   726→                poll.closed && "cursor-default opacity-90"
   727→              )}
   728→            >
   729→              {/* Progress fill (za tekstom, širina glede na zmagovalca) */}
   730→              <span
   731→                aria-hidden="true"
   732→                className={cn(
   733→                  "absolute inset-y-0 left-0 transition-[width] duration-500",
   734→                  isMyVote ? "bg-primary/15" : "bg-muted"
   735→                )}
   736→                style={{ width: `${widthPct}%` }}
   737→              />
   738→
   739→              <span className="relative z-10 flex min-w-0 flex-1 items-center gap-2">
   740→                <span
   741→                  className={cn(
   742→                    "flex size-5 shrink-0 items-center justify-center rounded-full border text-[10px] font-bold",
   743→                    isMyVote
   744→                      ? "border-primary bg-primary text-primary-foreground"
   745→                      : "border-border text-muted-foreground"
   746→                  )}
   747→                  aria-hidden="true"
   748→                >
   749→                  {isMyVote ? (
   750→                    <Check className="size-3" aria-hidden="true" />
   751→                  ) : (
   752→                    String.fromCharCode(65 + i)
   753→                  )}
   754→                </span>
   755→                <span className="min-w-0 truncate">{opt}</span>
   756→              </span>
   757→
   758→              <span className="relative z-10 flex shrink-0 items-center gap-2 text-xs">
   759→                {isPending && (
   760→                  <Loader2
   761→                    className="size-3.5 animate-spin text-primary"
   762→                    aria-hidden="true"
   763→                  />
   764→                )}
   765→                <span className="text-muted-foreground" aria-hidden="true">
   766→                  {pct}%
   767→                </span>
   768→                <span
   769→                  className={cn(
   770→                    "min-w-6 text-right font-semibold tabular-nums",
   771→                    isMyVote ? "text-primary" : "text-foreground"
   772→                  )}
   773→                  aria-hidden="true"
   774→                >
   775→                  {count}
   776→                </span>
   777→                <span className="sr-only">
   778→                  {count} {slUnit(count, "glas", "glasa", "glasi", "glasov")}
   779→                </span>
   780→              </span>
   781→            </button>
   782→          );
   783→        })}
   784→      </div>
   785→
   786→      {/* Noga: skupno + avtorska dejanja */}
   787→      <div className="mt-3 flex flex-wrap items-center justify-between gap-2">
   788→        <span className="text-xs text-muted-foreground">
   789→          {votesLabel(total)}
   790→          {poll.myVote !== null && " · tvoj glas je štet"}
   791→        </span>
   792→        {poll.isAuthor && (
   793→          <span className="flex items-center gap-1.5">
   794→            <Button
   795→              type="button"
   796→              variant="outline"
   797→              size="sm"
   798→              onClick={onToggleClose}
   799→              disabled={actionPending}
   800→              className="h-8 gap-1.5 px-2.5 text-xs"
   801→            >
   802→              {actionPending ? (
   803→                <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />
   804→              ) : (
   805→                <Lock className="size-3.5" aria-hidden="true" />
   806→              )}
   807→              {poll.closed ? "Znova odpri" : "Zaključi"}
   808→            </Button>
   809→            <Button
   810→              type="button"
   811→              variant="outline"
   812→              size="sm"
   813→              onClick={onDelete}
   814→              disabled={actionPending}
   815→              aria-label="Izbriši anketo"
   816→              className="h-8 gap-1.5 px-2.5 text-xs text-destructive hover:text-destructive"
   817→            >
   818→              {actionPending ? (
   819→                <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />

   821→                <Trash2 className="size-3.5" aria-hidden="true" />
   822→              )}
   823→              Izbriši
   824→            </Button>
   825→          </span>
   826→        )}
   827→      </div>
   828→    </div>
   829→  );
   830→}
