    96→async function getSharedItinerary(shareId: string) {
    97→  if (!SHARE_ID_RE.test(shareId)) return null;
    98→
    99→  const saved = await db.savedItinerary.findUnique({
   100→    where: { shareId },
   101→    select: {
   102→      name: true,
   103→      itinerary: true,
   104→      views: true,
   105→      createdAt: true,
   106→      isPublic: true,
   107→      // TASK 28 (live-sync): verzija ob renderu — osnova za polling banner.
   108→      contentVersion: true,
   109→      // ISSUE #8 §26 / F2-C (fork skupnostne poti): vhodni podatki
   110→      // načrtovalnika (PlannerInput) — podlaga za „Shrani kot svojo kopijo“.
   111→      formData: true,
   112→    },
   113→  });
   114→
   115→  if (!saved) return null;
   116→
   117→  // Parse itinererja — ob pokvarjenem JSON-u se obnašamo kot da ne obstaja
   118→  let itinerary: Itinerary;
   119→  try {
   120→    itinerary = JSON.parse(saved.itinerary) as Itinerary;
   121→  } catch {
   122→    console.error("[pot] pokvarjen JSON itinererja:", shareId);
   123→    return null;
   124→  }
   125→
   126→  if (!Array.isArray(itinerary?.days) || itinerary.days.length === 0) {
   127→    return null;
   128→  }
   129→
   130→  // Hardening: obdrži samo dneve z veljavnim seznamom lokacij
   131→  itinerary.days = itinerary.days.filter(
   132→    (d) => typeof d === "object" && d !== null && Array.isArray(d.locations)
   133→  );
   134→  if (itinerary.days.length === 0) return null;
   135→
   136→  // ISSUE #8 §26 / F2-C: formData (PlannerInput) za „Shrani kot svojo
   137→  // kopijo“. Starejše/anonimne pote imajo zapisan literal "null" (API:
   138→  // b.formData ?? null) → null; pokvarjen/veljaven le delno JSON → null
   139→  // (NE sesuje strani — fork takrat shrani kopijo brez formData).
   140→  let formData: PlannerInput | null = null;
   141→  if (typeof saved.formData === "string" && saved.formData !== "null") {
   142→    try {
   143→      const parsed: unknown = JSON.parse(saved.formData);
   144→      if (
   145→        parsed !== null &&
   146→        typeof parsed === "object" &&
   147→        !Array.isArray(parsed)
   148→      ) {
   149→        formData = parsed as PlannerInput;
   150→      }
   151→    } catch {
   152→      console.error("[pot] pokvarjen JSON formData:", shareId);
   153→    }
   154→  }
   155→
   156→  return { ...saved, itinerary, formData };
   157→}
