   414→function NavButton({
   415→  card,
   416→  lang,
   417→  variant = "hero",
   418→  origin,
   419→}: {
   420→  card: GoEntryCard;
   421→  lang: "sl" | "en";
   422→  variant?: "hero" | "icon";
   423→  /** ISSUE #4 §8: živi GPS — prenese se v web URL (external app dobi
   424→   * dejansko izhodišče; brez njega uporabi svojo lokacijo). */
   425→  origin?: { lat: number; lng: number } | null;
   426→}) {
   427→  const links = goNavLinks(card.entry, origin ?? null);
   428→  if (links == null) return null; // brez geo → handoff preprosto NI
   429→
   430→  const onNav = (e: React.MouseEvent<HTMLAnchorElement>) => {
   431→    // Mobilni: geo: URI odpre izbirnik aplikacij (precej nad web URL — brez
   432→    // privzganja Google Maps). Prestrežemo SAMO ob kliku (0 hidracijskih
   433→    // posledic). Desktop/pad: privzeti <a> odpre Google Maps.
   434→    if (isCoarsePointer()) {
   435→      e.preventDefault();
   436→      window.location.href = links.geo;
   437→    }
   438→  };
   439→
   440→  const externalHint = GO_NAV_LABELS.external[lang];
   441→  const aria = GO_NAV_LABELS.navigateAria[lang](card.entry.title);
   442→
   443→  if (variant === "icon") {
   444→    return (
   445→      <Button asChild variant="outline" className="h-11 w-11 shrink-0">
   446→        <a
   447→          href={links.web}
   448→          target="_blank"
   449→          rel="noopener noreferrer"
   450→          onClick={onNav}
   451→          title={externalHint}
